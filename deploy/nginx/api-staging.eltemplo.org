# Fuente de verdad: el server. Este archivo es su espejo; nginx no se despliega por pipeline.
# Staging API Reverse Proxy (port 4001)
# Location: /etc/nginx/sites-available/api-staging.eltemplo.org
# SSL will be added by certbot --nginx

server {
    server_name api-staging.eltemplo.org;

    # Proxy ALL requests to Fastify (routes are /api/*, /health)
    location / {
        proxy_pass http://127.0.0.1:4001;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Pass original client info
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Timeouts
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;

        # Don't buffer responses
        proxy_buffering off;

        # Retry on upstream restart window (pm2 reload, deploy). Only GET/HEAD by default.
        proxy_next_upstream error timeout http_502 http_503 http_504;
        proxy_next_upstream_tries 3;
        proxy_next_upstream_timeout 10s;
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;

    listen [::]:443 ssl; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/api-staging.eltemplo.org/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/api-staging.eltemplo.org/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}


server {
    if ($host = api-staging.eltemplo.org) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    listen 80;
    listen [::]:80;
    server_name api-staging.eltemplo.org;
    return 404; # managed by Certbot


}
