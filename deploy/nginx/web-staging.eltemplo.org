# Staging Landing Page (SSG)
# Location: /etc/nginx/sites-available/web-staging.eltemplo.org
# SSL will be added by certbot --nginx
# Etapa 1 de noindex (D-09/D-26): X-Robots-Tag noindex SIN bloquear robots.txt,
# para que Google pueda rastrear, ver el noindex y desindexar. La etapa 2
# (bloqueo en robots.txt) esta en el runbook, semanas despues.
# Paridad con eltemplo.org: 404 real y redirects reales para probar antes de prod.
# ATENCION: este archivo NO tiene las lineas ssl_certificate* que certbot agrego
# en el servidor. Aplicar con deploy/RUNBOOK-195-NGINX.md (merge de 3 vias),
# nunca copiarlo encima del live (rompe TLS).

server {
    listen 80;
    listen [::]:80;
    server_name web-staging.eltemplo.org;

    root /var/www/staging/el-templo-web;
    index index.html;

    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_min_length 256;
    gzip_comp_level 6;
    gzip_types
      text/plain
      text/css
      text/javascript
      application/javascript
      application/x-javascript
      application/json
      application/xml
      image/svg+xml
      font/woff2;

    # 404 real (fase 195, D-05): una ruta que no existe responde 404 con el shell
    # de 404.html (Nuxt). NO volver a agregar fallback a /index.html: toda pagina
    # debe estar prerenderizada.
    error_page 404 /404.html;
    location = /404.html {
        internal;
    }

    # Redirects reales (antes eran meta-refresh). Con y sin barra final.
    # Los routeRules de nuxt.config.ts quedan como respaldo.
    location ~ ^/curso-entrenadores/?$ {
        return 301 /academy;
    }
    location ~ ^/qr/bcn/?$ {
        return 302 https://app.eltemplo.org/register?promo=TEMPLOPASSBCN;
    }
    location ~ ^/qr/aura-club/?$ {
        return 302 https://app.eltemplo.org/register?promo=AURACLUB1;
    }

    location / {
        try_files $uri $uri/index.html $uri.html =404;
    }

    # Cache static assets aggressively (Nuxt/Vite adds content hashes)
    # Define su propio add_header: no hereda los del server, se repiten (Pitfall 3).
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot|webp|avif)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header X-Robots-Tag "noindex, nofollow" always;
        add_header X-Content-Type-Options "nosniff" always;
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header X-Robots-Tag "noindex, nofollow" always;
}

# SSL + HTTP/2 (certbot will add certificate paths)
