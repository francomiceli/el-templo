# Staging Member App (SPA)
# Location: /etc/nginx/sites-available/app-staging.eltemplo.org
# CRITICAL: try_files is REQUIRED for history mode - without it, direct URL access returns 404
# SSL will be added by certbot --nginx
# Etapa 1 de noindex (D-09/D-26): X-Robots-Tag noindex SIN bloquear robots.txt
# (la etapa 2 esta en deploy/RUNBOOK-195-NGINX.md).
# ATENCION: este archivo NO tiene las lineas ssl_certificate* que certbot agrego
# en el servidor. Aplicar con deploy/RUNBOOK-195-NGINX.md (merge de 3 vias),
# nunca copiarlo encima del live (rompe TLS).

server {
    listen 80;
    listen [::]:80;
    server_name app-staging.eltemplo.org;

    root /opt/el-templo-staging/app/spa;
    index index.html;

    # Deep links (D-14): se sirven como JSON, sin caer al index.html de la SPA.
    # Cada location define su propio add_header: se repiten los de seguridad.
    location = /.well-known/apple-app-site-association {
        default_type application/json;
        add_header X-Robots-Tag "noindex, nofollow" always;
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        try_files $uri =404;
    }
    location = /.well-known/assetlinks.json {
        default_type application/json;
        add_header X-Robots-Tag "noindex, nofollow" always;
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        try_files $uri =404;
    }

    # SPA fallback - all routes serve index.html, Vue Router handles the rest
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets aggressively (Vite adds content hashes)
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header X-Robots-Tag "noindex, nofollow" always;
        add_header X-Content-Type-Options "nosniff" always;
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Robots-Tag "noindex, nofollow" always;
}
