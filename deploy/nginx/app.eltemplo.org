# Member App (SPA - hash mode routing)
# Location: /etc/nginx/sites-available/app.eltemplo.org
# SSL will be added by certbot --nginx

server {
    listen 80;
    listen [::]:80;
    server_name app.eltemplo.org;

    root /var/www/member-app/spa;
    index index.html;

    # SPA fallback (defense-in-depth for hash mode)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets aggressively (Vite adds content hashes)
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
}
