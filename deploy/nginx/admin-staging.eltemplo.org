# Staging Admin App (SPA)
# Location: /etc/nginx/sites-available/admin-staging.eltemplo.org
# CRITICAL: try_files is REQUIRED for history mode - without it, direct URL access returns 404
# SSL will be added by certbot --nginx

server {
    listen 80;
    listen [::]:80;
    server_name admin-staging.eltemplo.org;

    root /opt/el-templo-staging/admin;
    index index.html;

    # SPA fallback - all routes serve index.html, Vue Router handles the rest
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets aggressively (Vite adds content hashes)
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
}
