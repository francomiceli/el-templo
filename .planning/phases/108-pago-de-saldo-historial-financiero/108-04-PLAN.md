    ```typescript
    const visibleConcepts = computed<OutstandingConcept[]>(() => {
      // D-21: Multi-currency anomaly. Si hay >1 currency, log warning + return solo la mayoritaria.
      const byCurrency = new Map<string, OutstandingConcept[]>();
      for (const c of props.outstandingConcepts) {
        if (!byCurrency.has(c.currency)) byCurrency.set(c.currency, []);
        byCurrency.get(c.currency)!.push(c);
      }
      if (byCurrency.size > 1) {
        log.warn('Multi-currency outstanding detectado', {
          memberId: props.userId,
          currencies: Array.from(byCurrency.keys()),
        });
        // mayoritaria = currency con más conceptos
        let majority = '';
        let max = 0;
        for (const [cur, arr] of byCurrency) {
          if (arr.length > max) { max = arr.length; majority = cur; }
        }
        return byCurrency.get(majority) ?? [];
      }
      return props.outstandingConcepts;
    });

    // Fallback 'ARS' es safe porque isInvalid bloquea Confirmar cuando visibleConcepts está vacío
    // (montoRecibido === null al abrir + auto-FIFO no aloca nada → sumAllocated === 0 → isInvalid === true).
    // El currency real se usa solo cuando hay al menos 1 concepto visible; fallback nunca llega al payload.
    const displayCurrency = computed(() => visibleConcepts.value[0]?.currency ?? 'ARS');

    const totalSaldos = computed(() => visibleConcepts.value.reduce((sum, c) => sum + c.balance, 0));

    const conceptKey = (c: OutstandingConcept) => `${c.targetKind}:${c.targetId}`;

    const sumAllocated = computed(() =>
      visibleConcepts.value.reduce((sum, c) => sum + (allocations.value[conceptKey(c)] ?? 0), 0),
    );

    // D-09/D-10/D-11: Confirmar disabled cuando Σ ≠ montoRecibido o montoRecibido === 0/null/empty-string.
    // v-model.number deja `null` cuando el input está vacío en algunos browsers, pero también puede dejar
    // string vacío o NaN — guard explícito contra los 3 casos.
    const isInvalid = computed(() => {
      if (
        montoRecibido.value === null ||
        typeof montoRecibido.value !== 'number' ||
        !Number.isFinite(montoRecibido.value) ||
        montoRecibido.value <= 0
      ) {
        return true;
      }
      if (sumAllocated.value !== montoRecibido.value) return true;
      return false;
    });

    const sumStatus = computed<'ok' | 'short' | 'over'>(() => {
      if (montoRecibido.value === null) return 'short';
      if (sumAllocated.value === montoRecibido.value) return 'ok';
      if (sumAllocated.value < montoRecibido.value) return 'short';
      return 'over';
    });

    const sumDiff = computed(() => Math.abs((montoRecibido.value ?? 0) - sumAllocated.value));
    ```

    **Auto-FIFO greedy (per 108-PATTERNS §8):**
    ```typescript
    function autoFifoAllocate(amount: number): void {
      let remaining = amount;
      const next: Record<string, number> = {};
      for (const c of visibleConcepts.value) {  // already FIFO from backend
        if (remaining <= 0) {
          next[conceptKey(c)] = 0;
          continue;
        }
        const allocated = Math.min(c.balance, remaining);
        next[conceptKey(c)] = allocated;
        remaining -= allocated;
      }
      allocations.value = next;
    }

    // D-07: re-ejecutar auto-FIFO cuando cambia montoRecibido.
    watch(montoRecibido, (val) => {
      if (val === null) {
        // Reset allocations a 0 cuando se borra el monto.
        for (const c of visibleConcepts.value) allocations.value[conceptKey(c)] = 0;
        return;
      }
      autoFifoAllocate(val);
    });

    // Auto-FIFO al abrir el dialog (cuando modelValue passes false→true).
    const defaultPaymentMethod: PaymentMethod = PAYMENT_METHOD_OPTIONS[0]?.value ?? 'cash';

    watch(() => props.modelValue, (open) => {
      if (open) {
        // Reset state.
        montoRecibido.value = null;
        paymentMethod.value = defaultPaymentMethod;
        fecha.value = todayISO();
        notes.value = '';
        const init: Record<string, number> = {};
        for (const c of visibleConcepts.value) init[conceptKey(c)] = 0;
        allocations.value = init;
        submitting.value = false;
      }
    });

    // Botón "Pagar todo" (D-07).
    function pagarTodo() {
      montoRecibido.value = totalSaldos.value;
      // watch will trigger autoFifoAllocate.
    }
    ```

    ```typescript
    async function onConfirm() {
      if (isInvalid.value) return;
      submitting.value = true;
      try {
        // D-08: filter out allocations === 0 (no enviar links de $0 — service rechaza).
        const links = visibleConcepts.value
          .map((c) => ({
            targetKind: c.targetKind,
            targetId: c.targetId,
            allocatedAmount: allocations.value[conceptKey(c)] ?? 0,
          }))
          .filter((l) => l.allocatedAmount > 0);

        if (links.length === 0) {
          $q.notify({ type: 'negative', message: 'No hay conceptos con monto asignado' });
          return;
        }

        const payload: RegisterPaymentInput = {
          memberId: props.userId,
          kind: 'debt_settlement',
          direction: 'inflow',
          amount: montoRecibido.value!,
          currency: displayCurrency.value,
          paymentMethod: paymentMethod.value,
          transactionDate: fecha.value,
          effectiveDate: fecha.value,  // D-22 implícito: same as transactionDate por default. Si el spec del backend pide otra cosa, ajustar.
          branchId: props.memberBranchId,
          notes: notes.value.trim() || null,
          links,
        };

        await transactionsApi.createTransaction(payload);
        $q.notify({ type: 'positive', message: 'Pago registrado correctamente' });
        emit('paid');
        emit('update:modelValue', false);
      } catch (err: unknown) {
        const message = err instanceof Error ? err.message : 'Error desconocido';
        log.error('Error registrando pago', { error: message });
        $q.notify({ type: 'negative', message });
      } finally {
        submitting.value = false;
      }
    }
    ```
