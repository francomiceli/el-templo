import type { CoachExampleBlock, BlockComparison, ValidationIssue } from './types';
import type { BlockRole, LevelGroup } from '../types';

const DIFFICULTY_TOLERANCE = 0.5; // DIFF-05: +/-0.5 tolerance

export function compareBlockToExample(
  expected: CoachExampleBlock,
  actual: {
    route: string;
    intensity: number;
    exercises: { name: string; contraction: string; dificultadLineal: number }[];
  } | null
): BlockComparison {
  const issues: ValidationIssue[] = [];

  if (!actual) {
    return {
      ...baseComparison(expected),
      actual: null,
      issues: [{ field: 'generation', expected: 'success', actual: 'failed', severity: 'error' }],
      passed: false,
    };
  }

  // Compare route
  if (expected.route !== actual.route) {
    issues.push({
      field: 'route',
      expected: expected.route,
      actual: actual.route,
      severity: 'error',
    });
  }

  // Compare intensity (ALGO-03)
  if (expected.intensity !== actual.intensity) {
    issues.push({
      field: 'intensity',
      expected: expected.intensity,
      actual: actual.intensity,
      severity: 'error',
    });
  }

  // Compare exercise count (BLOCK-06)
  const expectedCount = expected.exercises.length;
  const actualCount = actual.exercises.length;
  if (expectedCount !== actualCount) {
    issues.push({
      field: 'exerciseCount',
      expected: expectedCount,
      actual: actualCount,
      severity: 'warning',
    });
  }

  // Compare average difficulty (DIFF-05)
  const expectedAvg = calculateAvgDifficulty(expected.exercises);
  const actualAvg = calculateAvgDifficulty(actual.exercises);
  if (Math.abs(expectedAvg - actualAvg) > DIFFICULTY_TOLERANCE) {
    issues.push({
      field: 'avgDifficulty',
      expected: expectedAvg,
      actual: actualAvg,
      tolerance: DIFFICULTY_TOLERANCE,
      severity: 'warning',
    });
  }

  // Compare contraction distribution (ALGO-04)
  const expectedMix = getContractionMix(expected.exercises);
  const actualMix = getContractionMix(actual.exercises);
  if (!contractionMixesMatch(expectedMix, actualMix)) {
    issues.push({
      field: 'contractionMix',
      expected: expectedMix,
      actual: actualMix,
      severity: 'error',
    });
  }

  return {
    week: expected.week,
    day: expected.day,
    levelGroup: expected.levelGroup,
    role: expected.role,
    expected: {
      route: expected.route,
      intensity: expected.intensity,
      exerciseCount: expectedCount,
      avgDifficulty: expectedAvg,
      contractionMix: expectedMix,
    },
    actual: {
      route: actual.route,
      intensity: actual.intensity,
      exerciseCount: actualCount,
      avgDifficulty: actualAvg,
      contractionMix: actualMix,
    },
    issues,
    passed: issues.filter(i => i.severity === 'error').length === 0,
  };
}

function calculateAvgDifficulty(exercises: { difficulty?: number; dificultadLineal?: number }[]): number {
  const difficulties = exercises.map(e => e.dificultadLineal ?? e.difficulty ?? 0);
  return difficulties.reduce((a, b) => a + b, 0) / difficulties.length;
}

function getContractionMix(exercises: { contraction: string }[]): { CON: number; EXC: number; ISO: number } {
  return {
    CON: exercises.filter(e => e.contraction === 'CON' || e.contraction === 'CON.').length,
    EXC: exercises.filter(e => e.contraction === 'EXC' || e.contraction === 'EXC.').length,
    ISO: exercises.filter(e => e.contraction === 'ISO' || e.contraction === 'ISO.').length,
  };
}

function contractionMixesMatch(a: { CON: number; EXC: number; ISO: number }, b: { CON: number; EXC: number; ISO: number }): boolean {
  return a.CON === b.CON && a.EXC === b.EXC && a.ISO === b.ISO;
}
