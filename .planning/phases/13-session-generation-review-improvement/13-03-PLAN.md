```typescript
import * as fs from 'fs';
import * as path from 'path';
import { parse } from 'fast-csv';
import type { CoachExampleBlock } from './types';

const EXAMPLES_DIR = path.resolve(__dirname, '../../../../docs/session-logic/examples');

// CSV columns from examples:
// Nivel,Bloque,Bloque_nivel,Enfoque,Ruta,Patron,Bloque 2,Block id,Ejercicio,Esfuerzo,Dificultad,Repeticiones,Formato,Sets,Descanso (segs)

interface CsvRow {
  Nivel: string;
  Bloque: string;
  Bloque_nivel: string;
  Enfoque: string;
  Ruta: string;
  Patron: string;
  'Bloque 2': string;
  'Block id': string;
  Ejercicio: string;
  Esfuerzo: string;
  Dificultad: string;
  Repeticiones: string;
  Formato: string;
  Sets: string;
  'Descanso (segs)': string;
}

export async function parseCoachExamples(
  weekStart: number = 3,
  weekEnd: number = 21
): Promise<CoachExampleBlock[]> {
  const blocks: CoachExampleBlock[] = [];

  for (let week = weekStart; week <= weekEnd; week++) {
    const files = fs.readdirSync(EXAMPLES_DIR)
      .filter(f => f.includes(`Semana ${week} `) && !f.includes('Planificación Final') && !f.includes('Semana.csv') && f.endsWith('.csv'));

    for (const file of files) {
      const day = extractDayFromFilename(file);
      const filePath = path.join(EXAMPLES_DIR, file);
      const dayBlocks = await parseCoachDayFile(filePath, week, day);
      blocks.push(...dayBlocks);
    }
  }

  return blocks;
}

function extractDayFromFilename(filename: string): string {
  const match = filename.match(/- (Lunes|Martes|Miercoles|Jueves|Viernes|Sabado)\.csv$/);
  return match ? match[1] : 'Unknown';
}

async function parseCoachDayFile(
  filePath: string,
  week: number,
  day: string
): Promise<CoachExampleBlock[]> {
  return new Promise((resolve, reject) => {
    const rows: CsvRow[] = [];

    fs.createReadStream(filePath)
      .pipe(parse({ headers: true }))
      .on('data', (row: CsvRow) => rows.push(row))
      .on('end', () => {
        const blocks = groupRowsIntoBlocks(rows, week, day);
        resolve(blocks);
      })
      .on('error', reject);
  });
}

function groupRowsIntoBlocks(rows: CsvRow[], week: number, day: string): CoachExampleBlock[] {
  // Group rows by (memberLevel, role) to form blocks
  const blockMap = new Map<string, CsvRow[]>();

  for (const row of rows) {
    if (!row.Nivel || !row.Bloque || !row.Ejercicio) continue;

    const key = `${row.Nivel.toLowerCase()}-${row.Bloque}`;
    if (!blockMap.has(key)) {
      blockMap.set(key, []);
    }
    blockMap.get(key)!.push(row);
  }

  // Convert grouped rows to CoachExampleBlock structures
  const blocks: CoachExampleBlock[] = [];

  for (const [key, blockRows] of blockMap) {
    const firstRow = blockRows[0];
    const memberLevel = firstRow.Nivel.toLowerCase() as CoachExampleBlock['memberLevel'];

    // Map memberLevel to levelGroup
    const levelGroup = mapLevelToGroup(memberLevel);

    // Parse contraction (Esfuerzo): "CON.", "EXC.", "ISO." -> "CON", "EXC", "ISO"
    const exercises = blockRows.map(row => ({
      name: row.Ejercicio,
      contraction: row.Esfuerzo.replace('.', '').toUpperCase() as 'CON' | 'EXC' | 'ISO',
      difficulty: parseDifficulty(row.Dificultad, memberLevel),
    }));

    // Calculate intensity from SPOM lookup (week + route)
    // For validation, we'll use a placeholder; actual intensity comes from SPOM table
    const intensity = 0; // Will be resolved from SPOM during comparison

    blocks.push({
      week,
      day,
      levelGroup,
      memberLevel,
      role: firstRow.Bloque,
      route: firstRow.Ruta || 'INITIUM',
      intensity,
      format: firstRow.Formato || 'Straight Sets',
      exercises,
    });
  }

  return blocks;
}

function mapLevelToGroup(level: string): 'alfa_delta' | 'sigma' | 'omega' {
  switch (level) {
    case 'alfa':
    case 'delta':
      return 'alfa_delta';
    case 'sigma':
      return 'sigma';
    case 'omega':
    case 'spartan':
      return 'omega';
    default:
      return 'alfa_delta';
  }
}

function parseDifficulty(diffStr: string, memberLevel: string): number {
  // Parse difficulty string and convert to linear 1-12 scale
  // Input is relative difficulty (1-3), output is linear based on level
  const relativeDiff = parseInt(diffStr, 10) || 1;

  const levelBase: Record<string, number> = {
    alfa: 0,      // 1-3
    delta: 3,     // 4-6
    sigma: 6,     // 7-8
    omega: 8,     // 9-10
    spartan: 10,  // 11-12
  };

  const base = levelBase[memberLevel] || 0;
  return base + relativeDiff;
}

export { parseCoachDayFile, groupRowsIntoBlocks }; // Export for testing
```

```typescript
import type { CoachExampleBlock, BlockComparison, ValidationIssue } from './types';
import type { BlockRole, LevelGroup } from '../types';

const DIFFICULTY_TOLERANCE = 0.5; // DIFF-05: +/-0.5 tolerance

export function compareBlockToExample(
  expected: CoachExampleBlock,
  actual: {
    route: string;
    intensity: number;
    format: string;
    exercises: { name: string; contraction: string; dificultadLineal: number }[];
  } | null
): BlockComparison {
  const issues: ValidationIssue[] = [];

  if (!actual) {
    return {
      ...baseComparison(expected),
      actual: null,
      issues: [{ field: 'generation', expected: 'success', actual: 'failed', severity: 'error' }],
      passed: false,
    };
  }

  // Compare route (ALGO-02)
  if (expected.route !== actual.route) {
    issues.push({
      field: 'route',
      expected: expected.route,
      actual: actual.route,
      severity: 'error',
    });
  }

  // Compare intensity (ALGO-03)
  if (expected.intensity !== actual.intensity) {
    issues.push({
      field: 'intensity',
      expected: expected.intensity,
      actual: actual.intensity,
      severity: 'error',
    });
  }

  // Compare exercise count (BLOCK-06)
  const expectedCount = expected.exercises.length;
  const actualCount = actual.exercises.length;
  if (expectedCount !== actualCount) {
    issues.push({
      field: 'exerciseCount',
      expected: expectedCount,
      actual: actualCount,
      severity: 'warning',
    });
  }

  // Compare average difficulty (DIFF-05)
  const expectedAvg = calculateAvgDifficulty(expected.exercises);
  const actualAvg = calculateAvgDifficulty(actual.exercises);
  if (Math.abs(expectedAvg - actualAvg) > DIFFICULTY_TOLERANCE) {
    issues.push({
      field: 'avgDifficulty',
      expected: expectedAvg,
      actual: actualAvg,
      tolerance: DIFFICULTY_TOLERANCE,
      severity: 'warning',
    });
  }

  // Compare contraction distribution (ALGO-04)
  const expectedMix = getContractionMix(expected.exercises);
  const actualMix = getContractionMix(actual.exercises);
  if (!contractionMixesMatch(expectedMix, actualMix)) {
    issues.push({
      field: 'contractionMix',
      expected: expectedMix,
      actual: actualMix,
      severity: 'error',
    });
  }

  // Compare format (FORM-01, FORM-02)
  if (normalizeFormat(expected.format) !== normalizeFormat(actual.format)) {
    issues.push({
      field: 'format',
      expected: expected.format,
      actual: actual.format,
      severity: 'warning', // Warning because format variations may be acceptable
    });
  }

  return {
    week: expected.week,
    day: expected.day,
    levelGroup: expected.levelGroup,
    role: expected.role,
    expected: {
      route: expected.route,
      intensity: expected.intensity,
      exerciseCount: expectedCount,
      avgDifficulty: expectedAvg,
      contractionMix: expectedMix,
      format: expected.format,
    },
    actual: {
      route: actual.route,
      intensity: actual.intensity,
      exerciseCount: actualCount,
      avgDifficulty: actualAvg,
      contractionMix: actualMix,
      format: actual.format,
    },
    issues,
    passed: issues.filter(i => i.severity === 'error').length === 0,
  };
}

function baseComparison(expected: CoachExampleBlock): Omit<BlockComparison, 'actual' | 'issues' | 'passed'> {
  return {
    week: expected.week,
    day: expected.day,
    levelGroup: expected.levelGroup,
    role: expected.role,
    expected: {
      route: expected.route,
      intensity: expected.intensity,
      exerciseCount: expected.exercises.length,
      avgDifficulty: calculateAvgDifficulty(expected.exercises),
      contractionMix: getContractionMix(expected.exercises),
      format: expected.format,
    },
  };
}

function calculateAvgDifficulty(exercises: { difficulty?: number; dificultadLineal?: number }[]): number {
  const difficulties = exercises.map(e => e.dificultadLineal ?? e.difficulty ?? 0);
  return difficulties.reduce((a, b) => a + b, 0) / difficulties.length;
}

function getContractionMix(exercises: { contraction: string }[]): { CON: number; EXC: number; ISO: number } {
  return {
    CON: exercises.filter(e => e.contraction === 'CON' || e.contraction === 'CON.').length,
    EXC: exercises.filter(e => e.contraction === 'EXC' || e.contraction === 'EXC.').length,
    ISO: exercises.filter(e => e.contraction === 'ISO' || e.contraction === 'ISO.').length,
  };
}

function contractionMixesMatch(a: { CON: number; EXC: number; ISO: number }, b: { CON: number; EXC: number; ISO: number }): boolean {
  return a.CON === b.CON && a.EXC === b.EXC && a.ISO === b.ISO;
}

function normalizeFormat(format: string): string {
  // Normalize format names for comparison (case-insensitive, handle variations)
  return format.toLowerCase().trim().replace(/\s+/g, ' ');
}
```

2. Create run-validation.ts with explicit pipeline wiring:
```typescript
import { parseCoachExamples } from './parse-coach-examples';
import { compareBlockToExample } from './compare-algorithm';
import type { ValidationReport, BlockComparison, CoachExampleBlock } from './types';

// Import session generation pipeline
import { generateDaySession } from '../pipeline/index';
import type { DaySession, Block } from '../types';
import { db } from '../../../db';
import { spomService } from '../../spom/service';

export async function runValidation(
  weekStart: number = 3,
  weekEnd: number = 21
): Promise<ValidationReport> {
  const coachExamples = await parseCoachExamples(weekStart, weekEnd);
  const comparisons: BlockComparison[] = [];

  // Group examples by (week, day, levelGroup) for efficient generation
  const groupedExamples = groupExamples(coachExamples);

  for (const [key, blocks] of groupedExamples) {
    const { week, day, levelGroup } = parseKey(key);

    // Generate algorithm output for this session using the pipeline
    let generatedSession: DaySession | null = null;
    try {
      generatedSession = await generateDaySession({
        weekNumber: week,
        day,
        levelGroup,
        db,
      });
    } catch (error) {
      console.error(`Failed to generate session for ${key}:`, error);
    }

    // Compare each coach block to corresponding generated block
    for (const coachBlock of blocks) {
      // Resolve intensity from SPOM for coach block comparison
      const spomData = await spomService.lookup(week, coachBlock.route);
      const coachBlockWithIntensity = {
        ...coachBlock,
        intensity: spomData?.intensity ?? 0,
      };

      // Find corresponding generated block by role
      const generatedBlock = generatedSession?.blocks.find(
        b => normalizeRole(b.role) === normalizeRole(coachBlock.role)
      );

      // Extract comparable data from generated block
      const actualBlock = generatedBlock ? {
        route: generatedBlock.route,
        intensity: generatedBlock.intensity,
        format: generatedBlock.format?.name ?? 'Straight Sets',
        exercises: generatedBlock.prescriptions.map(p => ({
          name: p.exerciseName,
          contraction: p.contraction,
          dificultadLineal: p.dificultadLineal,
        })),
      } : null;

      const comparison = compareBlockToExample(coachBlockWithIntensity, actualBlock);
      comparisons.push(comparison);
    }
  }

  // Compile report
  const passedBlocks = comparisons.filter(c => c.passed).length;
  const issuesByType: Record<string, number> = {};

  for (const comp of comparisons) {
    for (const issue of comp.issues) {
      issuesByType[issue.field] = (issuesByType[issue.field] || 0) + 1;
    }
  }

  return {
    totalBlocks: comparisons.length,
    passedBlocks,
    failedBlocks: comparisons.length - passedBlocks,
    issuesByType,
    detailedComparisons: comparisons,
  };
}

function groupExamples(examples: CoachExampleBlock[]): Map<string, CoachExampleBlock[]> {
  const grouped = new Map<string, CoachExampleBlock[]>();

  for (const example of examples) {
    const key = `${example.week}-${example.day}-${example.levelGroup}`;
    if (!grouped.has(key)) {
      grouped.set(key, []);
    }
    grouped.get(key)!.push(example);
  }

  return grouped;
}

function parseKey(key: string): { week: number; day: string; levelGroup: 'alfa_delta' | 'sigma' | 'omega' } {
  const [weekStr, day, levelGroup] = key.split('-');
  return {
    week: parseInt(weekStr, 10),
    day,
    levelGroup: levelGroup as 'alfa_delta' | 'sigma' | 'omega',
  };
}

function normalizeRole(role: string): string {
  // Normalize role names for matching (handle variations)
  return role.toLowerCase().replace(/\s+/g, '_').replace('deuteros', 'deuteros');
}

// CLI runner
if (require.main === module) {
  console.log('Running validation suite...');

  runValidation(3, 21)
    .then(report => {
      console.log('\n=== VALIDATION REPORT ===\n');
      console.log(`Total blocks: ${report.totalBlocks}`);
      console.log(`Passed: ${report.passedBlocks} (${(report.passedBlocks / report.totalBlocks * 100).toFixed(1)}%)`);
      console.log(`Failed: ${report.failedBlocks}`);
      console.log('\nIssues by type:');
      for (const [type, count] of Object.entries(report.issuesByType)) {
        console.log(`  ${type}: ${count}`);
      }

      // Output detailed failures
      const failures = report.detailedComparisons.filter(c => !c.passed);
      if (failures.length > 0) {
        console.log('\n=== SAMPLE FAILURES (first 10) ===\n');
        for (const failure of failures.slice(0, 10)) {
          console.log(`Week ${failure.week}, ${failure.day}, ${failure.levelGroup}, ${failure.role}:`);
          for (const issue of failure.issues) {
            console.log(`  - ${issue.field}: expected ${JSON.stringify(issue.expected)}, got ${JSON.stringify(issue.actual)}`);
          }
        }
      }

      process.exit(report.failedBlocks > 0 ? 1 : 0);
    })
    .catch(error => {
      console.error('Validation failed:', error);
      process.exit(1);
    });
}
```
