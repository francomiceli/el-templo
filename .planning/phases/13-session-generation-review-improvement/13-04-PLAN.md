```typescript
import { exercises } from '../../../db/schema/exercises';
import { and, or, eq, inArray, like } from 'drizzle-orm';

async function selectInitiumExercises(
  ctx: InitiumContext,
  db: Database,
  nucleusRoute?: string
): Promise<SelectedExercise[]> {
  // Step 1: Try contextual selection if nucleusRoute provided
  if (nucleusRoute) {
    const relevantMobility = ROUTE_TO_MOBILITY[nucleusRoute] || [];

    if (relevantMobility.length > 0) {
      // Query exercises where mobilityRelated contains any of the relevant areas
      // The mobilityRelated column exists in schema (varchar(100), line 18 in exercises.ts)
      const contextualExercises = await db.select()
        .from(exercises)
        .where(and(
          or(
            eq(exercises.pattern, 'FLOW'),
            eq(exercises.category, 'Movilidad')
          ),
          or(
            ...relevantMobility.map(area =>
              like(exercises.mobilityRelated, `%${area}%`)
            )
          )
        ))
        .orderBy(exercises.id)
        .limit(ctx.exerciseCount);

      if (contextualExercises.length >= ctx.exerciseCount) {
        // Add trace for contextual selection
        return contextualExercises.map(e => ({
          exerciseId: e.id,
          name: e.exercise,
          contraction: 'ISO', // Default for warmup
          difficulty: e.difficulty,
          dificultadLineal: e.dificultadLineal ?? e.difficulty,
        }));
      }
      // If not enough contextual matches, fall through to generic selection
    }
  }

  // Step 2: Fallback to generic FLOW/Movilidad selection
  // ... existing generic selection logic
}
```
