         ```typescript
         fastify.post<{ Params: { userId: number }; Body: { filename: string } }>(
           "/:userId/photo/upload-url",
           { schema: uploadPhotoUrlSchema },
           async (request, reply) => {
             if (!fastify.r2) {
               return reply.code(503).send({ error: "Service Unavailable", message: "Image storage not configured" });
             }
             const sanitized = request.body.filename.toLowerCase().replace(/[^a-z0-9._-]/g, "-").replace(/-+/g, "-");
             const ext = sanitized.split(".").pop()?.toLowerCase() ?? "";
             const MIME_MAP: Record<string, string> = { jpg: "image/jpeg", jpeg: "image/jpeg", png: "image/png", webp: "image/webp" };
             const contentType = MIME_MAP[ext] ?? "image/jpeg";
             const key = `members/photos/${request.params.userId}-${Date.now()}.${ext || "jpg"}`;
             const command = new PutObjectCommand({ Bucket: fastify.r2Bucket, Key: key, ContentType: contentType });
             const uploadUrl = await getSignedUrl(fastify.r2, command, { expiresIn: 900 });
             const publicUrl = `${process.env.R2_PUBLIC_URL || ""}/${key}`;
             // Save publicUrl to DB
             await memberService.updatePhoto(request.params.userId, publicUrl);
             request.log.info({ userId: request.params.userId, key }, "Member photo upload URL generated");
             return { uploadUrl, publicUrl };
           }
         );
         ```
       Note: The route saves the publicUrl to DB immediately. The frontend then PUTs the file to the uploadUrl. The publicUrl is already stored, so if the upload fails the frontend can retry. This matches the blog image pattern.
