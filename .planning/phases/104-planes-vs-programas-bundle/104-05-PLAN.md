CRITICAL — view param resolution at the FE (per checker BLOCKER #1):

The earlier draft derived `view='program'` whenever `!hasPresencialPlan`, but for an online-only user with `currentProgramEnrollmentId === null` (the default after migration 0105) AND active enrollments, sending `view=program` results in 404 from the API ("No tenes un programa activo seleccionado"). R10 acceptance requires this user to enter Entrenar and see the weekly view of their program. The fix is to OMIT the `view` param in this exact scenario and let the server's first-enrollment fallback (Plan 03 resolveSessionView's last-resort branch) resolve.

Decision matrix for `params.view`:
| enrollmentId | hasPresencialPlan | enrollments.length | Action |
| ----------- | ----------------- | ----------------- | ------------------------------- |
| null | true | any | view='templo' |
| null | false | 0 | (user is blocked upstream) |
| null | false | >0 | OMIT view (server default) |
| not null | any | any | view='program' |

       import { computed } from 'vue'
       import { useUserStore } from 'src/stores/useUserStore'

       const log = createLogger('useCurrentProgram')

       /**
        * Phase 104: Composable for the program-selector UX in the weekly view.
        * Wraps useUserStore's currentProgram state and exposes a `select`
        * action that PUTs the new value and waits for confirmation.
        *
        * Per project convention this composable exposes cleanup() and does
        * NOT use onUnmounted internally.
        */
       export function useCurrentProgram() {
         const userStore = useUserStore()

         const currentEnrollmentId = computed(() => userStore.currentProgram.enrollmentId)
         const currentProgram = computed(() => userStore.currentProgram.program)
         const enrollments = computed(() => userStore.allActiveEnrollments)
         const hasPresencial = computed(() => userStore.hasPresencialPlan)
         const showSelector = computed(() => userStore.showProgramSelector)
         const isUpdating = computed(() => userStore.isUpdatingCurrentProgram)
         const viewKind = computed<'templo' | 'program'>(() =>
           currentEnrollmentId.value === null && hasPresencial.value ? 'templo' : 'program'
         )

         async function select(enrollmentId: number | null): Promise<void> {
           try {
             await userStore.setCurrentProgramId(enrollmentId)
           } catch (err: unknown) {
             log.error('Failed to set current program', { error: err instanceof Error ? err.message : String(err) })
             throw err
           }
         }

         function cleanup(): void {
           // No-op — this composable holds no subscriptions or timers.
         }

         return {
           currentEnrollmentId,
           currentProgram,
           enrollments,
           hasPresencial,
           showSelector,
           isUpdating,
           viewKind,
           select,
           cleanup,
         }
       }
