**RpeTrendChart.vue** - Line chart using vue-chart-3:
- Props: labels (string[]), data ((number | null)[])
- Register only needed Chart.js components (tree-shaking):
  - CategoryScale, LinearScale, PointElement, LineElement, Title, Tooltip, Filler
- Chart config:
  - Y axis: min 1, max 10, step 2 (RPE scale)
  - Line: bronze color (#b8956c), fill with 10% opacity
  - Points: navy background, bronze border
  - tension: 0.3 for smooth curve
  - spanGaps: true to connect across null values
  - responsive: true, maintainAspectRatio: false
  - Height: 200px
- Hide legend, show only chart

```vue
<template>
  <div class="rpe-chart-container">
    <LineChart
      :chart-data="chartData"
      :options="chartOptions"
      :height="200"
    />
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue';
import {
  Chart as ChartJS,
  CategoryScale,
  LinearScale,
  PointElement,
  LineElement,
  Title,
  Tooltip,
  Filler,
} from 'chart.js';
import { LineChart } from 'vue-chart-3';

// Register required components only (tree-shaking)
ChartJS.register(
  CategoryScale,
  LinearScale,
  PointElement,
  LineElement,
  Title,
  Tooltip,
  Filler
);

const props = defineProps<{
  labels: string[];
  data: (number | null)[];
}>();

const chartData = computed(() => ({
  labels: props.labels,
  datasets: [{
    label: 'RPE',
    data: props.data,
    fill: true,
    backgroundColor: 'rgba(184, 149, 108, 0.1)', // Bronze 10%
    borderColor: '#b8956c', // Bronze
    borderWidth: 2,
    pointBackgroundColor: '#2c3e5c', // Navy
    pointBorderColor: '#b8956c',
    pointRadius: 4,
    tension: 0.3,
    spanGaps: true,
  }],
}));

const chartOptions = {
  responsive: true,
  maintainAspectRatio: false,
  plugins: {
    legend: { display: false },
    title: { display: false },
  },
  scales: {
    y: {
      min: 1,
      max: 10,
      ticks: { stepSize: 2, color: '#8E8E8E' },
      grid: { color: 'rgba(44, 62, 92, 0.1)' },
    },
    x: {
      ticks: { color: '#8E8E8E', maxRotation: 45 },
      grid: { display: false },
    },
  },
};
</script>

<style scoped>
.rpe-chart-container {
  height: 200px;
  position: relative;
}
</style>
```

**EvaluationRequest.vue** - Shows eligibility and request button:
- Props: eligible (boolean), pending (boolean), averageRpe (number | null)
- Emits: request (when button clicked)
- States:
  1. Not eligible: Show message "Sigue entrenando para desbloquear evaluacion"
  2. Eligible, not pending: Show "Listo para evaluacion" + "Solicitar Evaluacion" button
  3. Pending: Show "Solicitud Pendiente" status with clock icon
- Button: bronze background, navy text, disabled during loading
- Use QBtn with loading prop for request submission

```vue
<template>
  <q-card class="evaluation-card">
    <q-card-section class="text-center">
      <!-- Not eligible -->
      <template v-if="!eligible && !pending">
        <q-icon name="trending_up" size="48px" color="grey-5" />
        <div class="text-h6 q-mt-md">Evaluacion de Nivel</div>
        <div class="text-body2 text-grey-7 q-mt-sm">
          Mantiene un RPE promedio de 6 o menos durante 2 semanas para solicitar evaluacion.
        </div>
        <div v-if="averageRpe !== null" class="text-caption q-mt-md">
          Tu promedio actual: {{ averageRpe.toFixed(1) }}
        </div>
      </template>

      <!-- Eligible, can request -->
      <template v-else-if="eligible && !pending">
        <q-icon name="emoji_events" size="48px" color="secondary" />
        <div class="text-h6 q-mt-md">Listo para Evaluacion</div>
        <div class="text-body2 q-mt-sm">
          Tu promedio RPE ({{ averageRpe?.toFixed(1) }}) indica que estas listo para avanzar.
        </div>
        <q-btn
          color="secondary"
          text-color="primary"
          label="Solicitar Evaluacion"
          class="q-mt-lg"
          :loading="loading"
          @click="onRequest"
        />
      </template>

      <!-- Pending request -->
      <template v-else-if="pending">
        <q-icon name="schedule" size="48px" color="secondary" />
        <div class="text-h6 q-mt-md">Solicitud Pendiente</div>
        <div class="text-body2 text-grey-7 q-mt-sm">
          Tu solicitud de evaluacion esta siendo revisada por tu coach.
        </div>
      </template>
    </q-card-section>
  </q-card>
</template>

<script setup lang="ts">
import { ref } from 'vue';

const props = defineProps<{
  eligible: boolean;
  pending: boolean;
  averageRpe: number | null;
}>();

const emit = defineEmits<{
  request: [];
}>();

const loading = ref(false);

async function onRequest() {
  loading.value = true;
  emit('request');
  // Parent will handle API call; loading cleared by prop changes
}
</script>

<style scoped lang="scss">
.evaluation-card {
  background: rgba(255, 255, 255, 0.95);
}
</style>
```
