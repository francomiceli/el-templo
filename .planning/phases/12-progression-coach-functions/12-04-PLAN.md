**MiCamino.vue** - Main progression page:
- Imports all 4 components
- Uses useProgressionStore() for state
- Uses useProgressionApi() for data fetching
- Calls fetchStats() on mount
- Shows loading spinner while fetching
- Error state with retry option
- Empty state for new users (no sessions yet)

Layout order:
1. LevelDisplay (top, centered)
2. TrainingStats (4 cards in grid)
3. QCard with "Tendencia de Esfuerzo" title + RpeTrendChart + average display
4. EvaluationRequest (bottom)

Pass appropriate props from store to each component.
Handle requestEvaluation event by calling API composable.

```vue
<template>
  <q-page padding class="mi-camino">
    <!-- Loading -->
    <q-spinner-dots
      v-if="loading"
      size="50px"
      color="primary"
      class="absolute-center"
    />

    <!-- Error -->
    <div v-else-if="error" class="text-center q-pa-xl">
      <q-icon name="error_outline" size="64px" color="negative" />
      <div class="text-h6 q-mt-md">Error al cargar datos</div>
      <q-btn
        label="Reintentar"
        color="primary"
        class="q-mt-md"
        @click="loadData"
      />
    </div>

    <!-- Empty state (no sessions) -->
    <div v-else-if="!stats || stats.totalSessions === 0" class="text-center q-pa-xl">
      <q-icon name="fitness_center" size="64px" color="grey-5" />
      <div class="text-h6 q-mt-md">Comienza tu camino</div>
      <div class="text-body2 text-grey-7 q-mt-sm">
        Completa tu primera sesion para ver tu progreso.
      </div>
      <q-btn
        label="Ir a Entrenar"
        color="primary"
        class="q-mt-lg"
        to="/training"
      />
    </div>

    <!-- Content -->
    <template v-else>
      <LevelDisplay
        :greek-letter="level?.greekLetter || ''"
        :level-name="level?.displayName || ''"
        class="q-mb-xl"
      />

      <TrainingStats :stats="stats" class="q-mb-lg" />

      <q-card class="q-mb-lg">
        <q-card-section>
          <div class="section-title">Tendencia de Esfuerzo (4 semanas)</div>
          <RpeTrendChart
            v-if="rpeTrend && rpeTrend.data.length > 0"
            :labels="rpeTrend.labels"
            :data="rpeTrend.data"
          />
          <div v-else class="text-center text-grey-6 q-py-xl">
            Completa sesiones para ver tu tendencia
          </div>
          <div
            v-if="rpeTrend && rpeTrend.averageRpe"
            class="text-center text-grey-7 q-mt-sm"
          >
            Promedio: {{ rpeTrend.averageRpe.toFixed(1) }}
          </div>
        </q-card-section>
      </q-card>

      <EvaluationRequest
        v-if="evaluation"
        :eligible="evaluation.eligible"
        :pending="evaluation.pendingRequest"
        :average-rpe="evaluation.averageRpeLast2Weeks"
        @request="handleEvaluationRequest"
      />
    </template>
  </q-page>
</template>

<script setup lang="ts">
import { computed, onMounted } from 'vue';
import { useProgressionStore } from '../stores/progressionStore';
import { useProgressionApi } from '../composables/useProgressionApi';
import LevelDisplay from '../components/LevelDisplay.vue';
import TrainingStats from '../components/TrainingStats.vue';
import RpeTrendChart from '../components/RpeTrendChart.vue';
import EvaluationRequest from '../components/EvaluationRequest.vue';

const progressionStore = useProgressionStore();
const { fetchStats, requestEvaluation, loading, error } = useProgressionApi();

const level = computed(() => progressionStore.level);
const stats = computed(() => progressionStore.stats);
const rpeTrend = computed(() => progressionStore.rpeTrend);
const evaluation = computed(() => progressionStore.evaluation);

async function loadData() {
  await fetchStats();
}

async function handleEvaluationRequest() {
  await requestEvaluation();
}

onMounted(() => {
  loadData();
});
</script>

  font-size: 1rem;
  color: $primary;
  letter-spacing: 0.05em;
  margin-bottom: 1rem;
}
</style>
```
