```typescript
import { mysqlTable, int, varchar, timestamp, mysqlEnum } from 'drizzle-orm/mysql-core';
import { users } from './users';

export const evaluationRequestStatus = mysqlEnum('evaluation_request_status', ['pending', 'approved', 'denied']);

export const evaluationRequests = mysqlTable('evaluation_requests', {
  id: int('id').primaryKey().autoincrement(),
  userId: int('user_id').notNull().references(() => users.id),
  requestedAt: timestamp('requested_at').defaultNow().notNull(),
  status: evaluationRequestStatus.default('pending').notNull(),
  averageRpeAtRequest: int('average_rpe_at_request'), // Snapshot at time of request
  processedAt: timestamp('processed_at'),
  processedBy: int('processed_by').references(() => users.id), // Coach who processed
  notes: varchar('notes', { length: 500 }), // Coach notes on decision
});
```

```typescript
// calculateStreak: Given sorted session dates (descending), count consecutive calendar days
// Streak breaks if most recent session is not today or yesterday
// Returns number of consecutive days

// Eligible if avgRpe <= 6 and avgRpe is not null
```

**routes.ts** - Two endpoints:

1. GET /progression/stats (authenticated):
   - Query user level from users table
   - Query completed_sessions for totalSessions, totalDaysTrained (COUNT DISTINCT date)
   - Query sessions in last 7 days for sessionsThisWeek
   - Query sessions for streak calculation (all sessions, sorted by date desc)
   - Query RPE data for last 4 weeks (date + rpe pairs)
   - Query average RPE for last 2 weeks (WHERE rpe IS NOT NULL)
   - Query evaluation_requests for pending request
   - Calculate eligibility (avgRpe <= 6)
   - Return combined response

2. POST /progression/request-evaluation (authenticated):
   - Check for existing pending request (reject if exists)
   - Verify eligibility (avgRpe <= 6 for last 2 weeks)
   - Insert into evaluation_requests with averageRpeAtRequest snapshot
   - Return success with requestId

Register routes in main index.ts under /api/progression prefix.
