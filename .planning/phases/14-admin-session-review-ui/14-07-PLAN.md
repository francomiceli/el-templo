Add endpoint to check weeks ahead coverage:

In admin service.ts:
```typescript
async getApprovedWeeksCoverage(): Promise<{
  currentWeek: number;
  weeksWithApproved: number[];
  weeksAhead: number;
}> {
  // Get current SPOM week
  const [configRow] = await this.db.select().from(schema.spomConfig);
  const currentWeek = configRow?.currentWeek || 1;

  // Get weeks that have at least one approved session
  const approvedWeeks = await this.db
    .selectDistinct({ week: schema.sessions.week })
    .from(schema.sessions)
    .where(
      and(
        eq(schema.sessions.status, 'approved'),
        gte(schema.sessions.week, currentWeek)
      )
    );

  const weeksWithApproved = approvedWeeks.map(r => r.week).sort((a, b) => a - b);
  const weeksAhead = weeksWithApproved.length > 0
    ? Math.max(...weeksWithApproved) - currentWeek
    : 0;

  return {
    currentWeek,
    weeksWithApproved,
    weeksAhead,
  };
}

// Auto-approve pending sessions for tomorrow
async autoApprovePendingSessions(): Promise<{ approved: number }> {
  // Get current SPOM week and determine tomorrow's day
  const [configRow] = await this.db.select().from(schema.spomConfig);
  const currentWeek = configRow?.currentWeek || 1;

  // Calculate tomorrow's day based on branch timezone
  const now = new Date();
  const tomorrow = new Date(now);
  tomorrow.setDate(tomorrow.getDate() + 1);

  const dayOfWeek = tomorrow.getDay(); // 0=Sun, 1=Mon, ...
  const dayMap: Record<number, string> = {
    1: 'lunes',
    2: 'martes',
    3: 'miercoles',
    4: 'jueves',
    5: 'viernes',
    6: 'sabado',
  };

  const tomorrowDay = dayMap[dayOfWeek];

  // If Sunday, skip (no sessions on Sunday)
  if (!tomorrowDay) {
    return { approved: 0 };
  }

  // Find all pending sessions for tomorrow
  const pendingSessions = await this.db
    .select()
    .from(schema.sessions)
    .where(
      and(
        eq(schema.sessions.status, 'pending_review'),
        eq(schema.sessions.week, currentWeek),
        eq(schema.sessions.day, tomorrowDay)
      )
    );

  if (pendingSessions.length === 0) {
    return { approved: 0 };
  }

  // Auto-approve them
  await this.db
    .update(schema.sessions)
    .set({
      status: 'approved',
      approvedAt: new Date(),
      approvedBy: null, // null indicates auto-approved (no user)
      autoApproved: true, // Flag to distinguish from manual approval
    })
    .where(
      and(
        eq(schema.sessions.status, 'pending_review'),
        eq(schema.sessions.week, currentWeek),
        eq(schema.sessions.day, tomorrowDay)
      )
    );

  return { approved: pendingSessions.length };
}
```

Create cron job file:
```typescript
// src/jobs/auto-approve.ts
import cron from 'node-cron';
import { AdminSessionService } from '../modules/admin/service';
import { db } from '../db';

export function startAutoApproveJob() {
  const adminService = new AdminSessionService(db);

  // Run at 23:59 every day (just before midnight)
  // This auto-approves sessions for the next day if not reviewed
  cron.schedule('59 23 * * *', async () => {
    console.log('[auto-approve] Running auto-approve job...');
    try {
      const result = await adminService.autoApprovePendingSessions();
      console.log(`[auto-approve] Auto-approved ${result.approved} sessions`);
    } catch (error) {
      console.error('[auto-approve] Error:', error);
    }
  }, {
    timezone: 'America/Argentina/Buenos_Aires', // Branch timezone
  });

  console.log('[auto-approve] Cron job scheduled for 23:59 daily');
}
```
