```ts
// el-templo-api/src/modules/platform/provisioning-service.ts
export interface ProvisionInput {
  name: string;                 // nombre del gimnasio
  slug: string;                 // subdominio, minúsculas
  country: string;              // ISO-2, p. ej. "AR"
  currency: string;             // ISO-3, p. ej. "ARS"
  timezone: string;             // p. ej. "America/Argentina/Buenos_Aires"
  brand?: { displayName?: string; primaryColor?: string; secondaryColor?: string; logoUrl?: string };
  physicalBranch?: { name: string; code: string; address?: string; maxCapacity?: number };
  owner: { email: string; firstName: string; lastName: string; password: string };
  actorPlatformUserId: number;  // sale del token verificado, NUNCA del body
}

export interface ProvisionResult {
  tenantId: number;
  slug: string;
  virtualBranchId: number;
  physicalBranchId: number | null;
  ownerUserId: number;
  moduleFlags: Array<{ key: string; value: string }>;   // 5 filas
  brandKeys: string[];                                   // solo las que difieren del default
}

export async function provisionTenant(db: Db, log: FastifyBaseLogger, input: ProvisionInput): Promise<ProvisionResult>;
export class SlugNoDisponibleError extends Error {}   // → 409 en la ruta
export class SlugInvalidoError extends Error {}       // → 400 en la ruta
```

`GET /api/platform/tenants/slug-disponible?slug=x` (200) responde `{ slug, disponible: boolean, motivo: "formato" | "reservado" | "tomado" | null }`.

| API → MySQL | Escritura multi-tabla que debe ser atómica y correctamente estampada por tenant |
| respuesta del alta → navegador | La contraseña del owner viaja una única vez |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-182-05 | Tampering | INSERTs del provisioner | mitigate | `tenantValues(ctx, …)` en los 4 grupos de INSERT (estampa el tenant DESPUÉS del spread ⇒ un `tenantId` del payload no puede ganar); test que afirma que ninguna fila queda con `tenant_id = 1` |
| T-182-06 | Denial of Service (datos) | Alta parcial | mitigate | Una sola `db.transaction`; validación de slug antes del primer INSERT; test de rollback que verifica cero filas |
| T-182-07 | Information Disclosure | Contraseña del owner | mitigate | Se devuelve una única vez en el 201; nunca se loguea (se loguea solo `{slug,tenantId}`), nunca entra al payload de auditoría (filtro explícito de claves `password*`), solo se persiste su hash argon2 |
| T-182-08 | Repudiation | Alta sin traza | mitigate | `platformAudit.write(tx, …)` dentro de la MISMA transacción, actor tomado de `request.platformUser`, nunca del body; test que verifica la fila y su `actor_id` |
| T-182-03b | Spoofing | Slug del gimnasio | mitigate | Validación de formato con regex anclada + `RESERVED_TENANT_SLUGS` + unique de DB; el chequeo previo evita que `ER_DUP_ENTRY` suba como 500 |
