User-level fields (`leadStatus`, `leadNotes`, `createdBy`, `converted`) come from the user row directly.

    **Query pattern (MySQL 8 — CTE preferred; if local MySQL is 5.7 use correlated subquery — verify by reading existing service methods):**

    ```sql
    -- D-03/D-42: pick the latest non-cancelled trial booking per user
    WITH latest_trial AS (
      SELECT b.member_id AS user_id,
             MAX(b.id)   AS booking_id  -- "latest" by id; equivalent to latest booking_date given the same user
      FROM bookings b
      WHERE b.is_trial = 1
        AND b.status <> 'cancelado'     -- D-43
      GROUP BY b.member_id
    )
    SELECT
      b.id           AS bookingId,
      u.id           AS userId,
      u.first_name, u.last_name,
      b.booking_date AS bookingDate,
      s.start_time   AS startTime,
      br.id          AS branchId,
      br.name        AS branchName,
      a.id           AS attendanceId,
      u.lead_status, u.lead_notes, u.converted_at,
      creator.id   AS creatorId,
      creator.first_name AS creatorFirstName,
      creator.last_name  AS creatorLastName
    FROM latest_trial lt
    JOIN bookings b      ON b.id = lt.booking_id
    JOIN users    u      ON u.id = lt.user_id
    JOIN schedules s     ON s.id = b.schedule_id
    JOIN branches  br    ON br.id = s.branch_id
    LEFT JOIN attendance a ON a.member_id   = b.member_id
                          AND a.schedule_id = b.schedule_id
                          AND a.session_date = b.booking_date
                          AND a.status = 'confirmado'
    LEFT JOIN users creator ON creator.id = u.created_by
    WHERE (br.country = ${country} OR br.is_virtual = 1)     -- D-24
      AND ...filters...                                       -- branch, date range, lead_status, attended, shift, gestionaUserId, daysWithoutConvertingMin, search
    ORDER BY b.booking_date DESC, b.id DESC
    LIMIT ${limit} OFFSET ${offset};
    ```
