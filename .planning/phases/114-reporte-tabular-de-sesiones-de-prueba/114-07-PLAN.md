    ```html
    <q-card v-if="memberProfile?.status === 'prueba'" class="q-pa-md q-mb-md">
      <div class="text-h6 q-mb-sm">Datos de Lead</div>

      <q-select
        v-model="leadDraft.leadStatus"
        :options="LEAD_STATUS_OPTIONS"
        emit-value
        map-options
        label="Estado del Lead"
        outlined
        dense
        :loading="savingStatus"

      />

      <q-input
        v-model="leadDraft.leadNotes"
        type="textarea"
        label="Comentarios"
        outlined
        autogrow
        maxlength="2000"
        :loading="savingNotes"
        @blur="onSaveLeadNotes"
        class="q-mt-sm"
      />

      <div class="q-mt-sm text-caption text-grey-7">
        Gestiona: <strong>{{ memberProfile.createdBy?.name ?? '—' }}</strong>
      </div>
    </q-card>
    ```

    Script (`<script setup lang="ts">`):
    ```ts
    import { ref, watch } from "vue";
    import { useQuasar } from "quasar";
    import { useMembersApi } from "src/composables/useMembersApi";
    import { createLogger } from "src/utils/logger";

    const LEAD_STATUS_OPTIONS = [
      { value: "en_seguimiento", label: "En seguimiento" },
      { value: "cerrado", label: "Cerrado" },
      { value: "perdido", label: "Perdido" },
    ] as const;

    const log = createLogger("AlumnoDetail.DatosDeLead");
    const $q = useQuasar();
    const { updateLead } = useMembersApi();

    // Local draft mirrors memberProfile fields; reset on profile reload.
    const leadDraft = ref<{ leadStatus: "en_seguimiento" | "cerrado" | "perdido" | null; leadNotes: string | null }>({
      leadStatus: null,
      leadNotes: null,
    });
    const savingStatus = ref(false);
    const savingNotes = ref(false);

    watch(
      () => memberProfile.value,
      (p) => {
        if (p?.status === "prueba") {
          leadDraft.value.leadStatus = p.leadStatus ?? null;
          leadDraft.value.leadNotes = p.leadNotes ?? null;
        }
      },
      { immediate: true },
    );

    async function onSaveLeadStatus(newValue: "en_seguimiento" | "cerrado" | "perdido"): Promise<void> {
      if (!memberProfile.value) return;
      const previous = memberProfile.value.leadStatus;
      savingStatus.value = true;
      try {
        const snapshot = await updateLead(memberProfile.value.userId, { leadStatus: newValue });
        memberProfile.value.leadStatus = snapshot.leadStatus;
        // D-34 invariant: leadNotes is NOT touched by lead_status edits server-side.
        // memberProfile.leadNotes stays as-is.
        $q.notify({ type: "positive", message: "Estado actualizado" });
      } catch (err: unknown) {
        leadDraft.value.leadStatus = previous;
        const message = err instanceof Error ? err.message : "Error al actualizar estado";
        log.error("Failed to update lead status", { err });
        $q.notify({ type: "negative", message });
      } finally {
        savingStatus.value = false;
      }
    }

    async function onSaveLeadNotes(): Promise<void> {
      if (!memberProfile.value) return;
      const draft = leadDraft.value.leadNotes;
      if (draft === memberProfile.value.leadNotes) return;
      savingNotes.value = true;
      try {
        const snapshot = await updateLead(memberProfile.value.userId, {
          leadNotes: draft === "" ? null : draft,
        });
        memberProfile.value.leadNotes = snapshot.leadNotes;
        $q.notify({ type: "positive", message: "Comentario guardado" });
      } catch (err: unknown) {
        leadDraft.value.leadNotes = memberProfile.value.leadNotes;
        const message = err instanceof Error ? err.message : "Error al guardar comentario";
        log.error("Failed to update lead notes", { err });
        $q.notify({ type: "negative", message });
      } finally {
        savingNotes.value = false;
      }
    }
    ```

    **CLAUDE.md compliance:** no `console.*`, no `any`. `err: unknown` + instanceof Error narrowing.

    Place this block in the page wherever makes visual sense — read the existing layout first; do NOT push it above the primary alumno header. A reasonable location is right under the existing `status === 'prueba'` banner (line ~132).
