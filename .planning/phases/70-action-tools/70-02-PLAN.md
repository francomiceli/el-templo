**Export pendingActions and a resolvePendingAction helper from tools.ts:**

```typescript
export function resolvePendingAction(
  phone: string,
): { tool: string; args: Record<string, unknown> } | undefined {
  const action = pendingActions.get(phone);
  if (action) pendingActions.delete(phone);
  return action;
}
```

2. **Add interactive button reply dispatch BEFORE AI processing.** At the top of the message handling flow in handler.ts, check if the incoming message has an `interactiveReplyId`. If so, dispatch directly to the pending action without going through the AI:

```typescript
import { resolvePendingAction } from "../ai/tools.js";

// In handleIncomingMessage, after parsing the message:
if (parsed.interactiveReplyId) {
  const replyId = parsed.interactiveReplyId;

  // Confirmation buttons
  if (replyId === "confirm_booking" || replyId === "confirm_trial") {
    const pending = resolvePendingAction(phone);
    if (pending) {
      const result = await executeTool(
        pending.tool,
        pending.args,
        db,
        conversationId,
        { phone, clientState },
      );
      await sendTextMessage(phone, result);
      return; // Skip AI processing entirely
    }
  }

  // Cancel buttons
  if (replyId === "cancel_booking" || replyId === "cancel_trial") {
    resolvePendingAction(phone); // Clear pending action
    await sendTextMessage(
      phone,
      "No hay problema, si cambias de idea aca estoy!",
    );
    return;
  }

  // Class selection buttons (schedule_XX_YYYY-MM-DD format from alternatives)
  if (replyId.startsWith("schedule_")) {
    const parts = replyId.split("_");
    const scheduleId = parseInt(parts[1], 10);
    const date = parts[2];
    // Re-invoke book_class with the selected schedule (unconfirmed, so it will show confirmation buttons)
    const result = await executeTool(
      "book_class",
      { scheduleId, date },
      db,
      conversationId,
      { phone, clientState },
    );
    if (result !== "[BUTTONS_SENT]") {
      await sendTextMessage(phone, result);
    }
    return;
  }
}
```

This ensures button presses are handled deterministically without relying on AI text parsing -- per user decision.
