**SPEC §8 LOCKED model:**

- `balances.amount > 0` = miembro debe (outstanding obligation).
- `balances.amount = 0` = saldado.
- `balances.amount < 0` = saldo a favor (D-08).
- For `target_kind='subscription'`: the subscription's `pricePaid` IS the obligation. The `balances` row is `obligation - settlements` running tally.
- For `target_kind='debt_balance'`: the row must already exist (caller responsibility — Phase 105 only path is via void of debt_settlement whose original link was against a subscription, where the row already exists from the prior plan_charge).
- For `target_kind='transaction'`: skip cache write entirely (transaction-to-transaction links don't move balances cache).

**Lazy seed logic — two-step (NOT a single ON DUPLICATE KEY UPDATE), because seeded value depends on target lookup:**

```typescript
import { and, eq, sql } from "drizzle-orm";
import {
  balances,
  subscriptions,
  financialTransactions,
} from "../../db/schema";
import { BadRequestError } from "../shared/errors";

async applyDelta(
  tx: TxType,
  transaction: FinancialTransactionRow,
  links: TransactionLinkRow[],
  sign: 1 | -1,
): Promise<void> {
  for (const link of links) {
    // target_kind='transaction' has no balances cache effect
    if (link.targetKind === "transaction") continue;

    // Sign convention: inflow REDUCES outstanding (member paid); outflow INCREASES it (refund).
    // Multiply by `sign` (+1 for create, -1 for void).
    const delta = sign * (transaction.direction === "inflow"
      ? -link.allocatedAmount
      : +link.allocatedAmount);

    // Look up existing row (per the UNIQUE composite key)
    const existing = await tx
      .select()
      .from(balances)
      .where(
        and(
          eq(balances.memberId, transaction.memberId),
          eq(balances.targetKind, link.targetKind),
          eq(balances.targetId, link.targetId),
          eq(balances.currency, transaction.currency),
        ),
      )
      .limit(1);

    if (existing.length > 0) {
      // Apply delta only — no re-seeding
      await tx
        .update(balances)
        .set({
          amount: sql`${balances.amount} + ${delta}`,
          lastRecomputedAt: sql`NOW()`,
        })
        .where(eq(balances.id, existing[0].id));
      this.log.info(
        {
          balanceId: existing[0].id,
          memberId: transaction.memberId,
          targetKind: link.targetKind,
          targetId: link.targetId,
          delta,
        },
        "Balance delta applied (existing row)",
      );
      continue;
    }

    // Lazy seed — first time this target is touched
    let seedAmount = 0;
    if (link.targetKind === "subscription") {
      const [sub] = await tx
        .select({
          pricePaid: subscriptions.pricePaid,
          currency: subscriptions.currency,
        })
        .from(subscriptions)
        .where(eq(subscriptions.id, link.targetId))
        .limit(1);
      if (!sub) {
        throw new BadRequestError(
          `Subscription ${link.targetId} no encontrada al sembrar balance`,
        );
      }
      if (sub.currency !== transaction.currency) {
        throw new BadRequestError(
          `Moneda inconsistente: subscription es ${sub.currency}, transacción es ${transaction.currency}`,
        );
      }
      seedAmount = sub.pricePaid;
    }
    // For target_kind='debt_balance': seedAmount stays 0 — caller must have seeded it.
    // For target_kind='transaction': we already `continue`d above.

    const finalAmount = seedAmount + delta;
    await tx.insert(balances).values({
      memberId: transaction.memberId,
      targetKind: link.targetKind,
      targetId: link.targetId,
      currency: transaction.currency,
      amount: finalAmount,
      lastRecomputedAt: new Date(),
    });
    this.log.info(
      {
        memberId: transaction.memberId,
        targetKind: link.targetKind,
        targetId: link.targetId,
        seedAmount,
        delta,
        finalAmount,
      },
      "Balance row seeded (lazy)",
    );
  }
}
```

**Why two-step instead of `INSERT ... ON DUPLICATE KEY UPDATE`:** the SEED value depends on a per-target lookup (subscription's `pricePaid`). A single upsert can't express "seed = pricePaid IF NOT EXISTS, ELSE leave existing untouched and add delta". Inside `db.transaction(...)` the SELECT-then-INSERT/UPDATE pair is atomic relative to other writers on the row; the UNIQUE constraint catches the rare race (concurrent first-touch), and the caller can retry.

```typescript
return await this.db.transaction(async (tx) => {
  // 1. Validate invariants BEFORE any write.
  //    1a. Σ allocated_amount === amount (only when links non-empty).
  if (input.links.length > 0) {
    const sum = input.links.reduce((acc, l) => acc + l.allocatedAmount, 0);
    if (sum !== input.amount) {
      throw new BadRequestError(
        "La suma de los montos asignados no coincide con el monto de la transacción",
      );
    }
  } else if (input.kind !== "advance_payment" && input.kind !== "adjustment") {
    throw new BadRequestError(
      `La transacción de tipo '${input.kind}' requiere al menos un link`,
    );
  }

  //    1b. Member exists (FK probe via users).
  const memberExists = await tx
    .select({ id: users.id })
    .from(users)
    .where(eq(users.id, input.memberId))
    .limit(1);
  if (memberExists.length === 0)
    throw new NotFoundError("Miembro no encontrado");

  //    1c. Branch exists.
  const branchExists = await tx
    .select({ id: branches.id })
    .from(branches)
    .where(eq(branches.id, input.branchId))
    .limit(1);
  if (branchExists.length === 0)
    throw new NotFoundError("Sucursal no encontrada");

  //    1d. For each link, probe target_id exists in the matching table.
  for (const link of input.links) {
    let exists: { id: number }[];
    switch (link.targetKind) {
      case "subscription":
        exists = await tx
          .select({ id: subscriptions.id })
          .from(subscriptions)
          .where(eq(subscriptions.id, link.targetId))
          .limit(1);
        break;
      case "debt_balance":
        exists = await tx
          .select({ id: balances.id })
          .from(balances)
          .where(eq(balances.id, link.targetId))
          .limit(1);
        break;
      case "transaction":
        exists = await tx
          .select({ id: financialTransactions.id })
          .from(financialTransactions)
          .where(eq(financialTransactions.id, link.targetId))
          .limit(1);
        break;
      default:
        throw new BadRequestError(
          `target_kind desconocido: ${String(link.targetKind)}`,
        );
    }
    if (exists.length === 0) {
      throw new NotFoundError(
        `Concepto enlazado no existe: ${link.targetKind} ${link.targetId}`,
      );
    }
  }

  // 2. INSERT financial_transactions.
  const inserted = await tx.insert(financialTransactions).values({
    memberId: input.memberId,
    kind: input.kind,
    direction: input.direction,
    amount: input.amount,
    currency: input.currency ?? "ARS",
    paymentMethod: input.paymentMethod,
    transactionDate: input.transactionDate,
    effectiveDate: input.effectiveDate,
    branchId: input.branchId,
    recordedBy,
    notes: input.notes ?? null,
  });
  const transactionId = Number(inserted[0].insertId);

  // 3. INSERT transaction_links (UNIQUE constraint catches dup target tuples → MySQL ER_DUP_ENTRY).
  if (input.links.length > 0) {
    await tx.insert(transactionLinks).values(
      input.links.map((l) => ({
        transactionId,
        targetKind: l.targetKind,
        targetId: l.targetId,
        allocatedAmount: l.allocatedAmount,
      })),
    );
  }

  // 4. Re-read the ledger row + links for the full detail (and for balanceService).
  const [txRow] = await tx
    .select()
    .from(financialTransactions)
    .where(eq(financialTransactions.id, transactionId));
  const linkRows = await tx
    .select()
    .from(transactionLinks)
    .where(eq(transactionLinks.transactionId, transactionId));

  // 5. Apply cache delta in the SAME tx (BalanceService handles lazy seed from subscription.pricePaid).
  await this.balanceService.applyDelta(tx, txRow, linkRows, +1);

  this.log.info(
    {
      transactionId,
      memberId: input.memberId,
      kind: input.kind,
      amount: input.amount,
      recordedBy,
      linkCount: linkRows.length,
    },
    "Financial transaction created",
  );

  return { ...txRow, links: linkRows };
});
```
