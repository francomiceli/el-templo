    **RPE message mapping (EXACT copy from D-12 and UI-SPEC):**
    ```typescript
    interface RpeMessage {
      icon: string
      iconColor: string
      background: string
      text: string
    }

    function getRpeMessage(rpe: number): RpeMessage {
      if (rpe <= 3) {
        return {
          icon: 'trending_up',
          iconColor: 'primary',
          background: 'rgba(192, 122, 86, 0.08)',  // rgba($primary, 0.08)
          text: 'Podrias subir la intensidad en tu proxima sesion',
        }
      }
      if (rpe <= 6) {
        return {
          icon: 'check_circle',
          iconColor: 'positive',
          background: 'rgba(90, 154, 107, 0.08)',  // rgba($positive, 0.08)
          text: 'Buen balance -- segui asi',
        }
      }
      if (rpe <= 8) {
        return {
          icon: 'hotel',
          iconColor: 'secondary',
          background: 'rgba(160, 117, 90, 0.08)',  // rgba($secondary, 0.08)
          text: 'Buen esfuerzo. Descansa bien hoy',
        }
      }
      // rpe 9-10
      return {
        icon: 'local_fire_department',
        iconColor: 'warning',
        background: 'rgba(212, 168, 67, 0.08)',  // rgba($warning, 0.08)
        text: 'Entrenaste al maximo. Considera un dia de recuperacion manana',
      }
    }
    ```

    **Computed:**
    ```typescript
    const message = computed(() => {
      if (!props.hasInteracted || props.rpeValue === null) return null
      return getRpeMessage(props.rpeValue)
    })
    ```
