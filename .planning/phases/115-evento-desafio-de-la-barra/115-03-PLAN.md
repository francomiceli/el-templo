```ts
export function useImageComposer() {
  /**
   * Composes user photo with frame via offscreen canvas.
   * Canvas: 1080x1920. Photo: object-fit cover, center-crop. Frame: drawn at full canvas size on top.
   * Export: image/jpeg @ quality 0.85.
   * @param photoBase64 - base64 data URL or raw base64 string from @capacitor/camera
   * @param framePath - public path to the frame PNG (e.g. '/desafio-barra/marco-placeholder.png')
   * @returns JPEG Blob ready for share or download
   */
  async function composeWithFrame(
    photoBase64: string,
    framePath: string,
  ): Promise<Blob>;
  return { composeWithFrame };
}
```

  <action>
    Crear `el-templo-app/src/modules/bar-challenge/composables/useImageComposer.ts`:
    - Import `createLogger` desde `src/utils/logger`; `const logger = createLogger('image-composer')`.
    - Export función `useImageComposer()` que retorna `{ composeWithFrame }`.
    - `async function composeWithFrame(photoBase64: string, framePath: string): Promise<Blob>`:
      1. Crear canvas offscreen: `const canvas = document.createElement('canvas'); canvas.width = 1080; canvas.height = 1920;`
      2. `const ctx = canvas.getContext('2d')`; si `!ctx`, `throw new Error('Canvas 2D no disponible')`.
      3. Helper interno `loadImage(src: string): Promise<HTMLImageElement>` que envuelve `new Image()` + `onload`/`onerror` en Promise.
      4. Asegurar que `photoBase64` tenga prefijo `data:image/...`; si llega como base64 raw (sin prefijo), antepponer `data:image/jpeg;base64,`.
      5. `const photo = await loadImage(photoBase64Normalized)`.
      6. Calcular cover-center crop:
         ```ts
         const targetW = 1080, targetH = 1920
         const scale = Math.max(targetW / photo.width, targetH / photo.height)
         const drawW = photo.width * scale
         const drawH = photo.height * scale
         const dx = (targetW - drawW) / 2
         const dy = (targetH - drawH) / 2
         ctx.drawImage(photo, dx, dy, drawW, drawH)
         ```
      7. `const frame = await loadImage(framePath); ctx.drawImage(frame, 0, 0, 1080, 1920)`.
      8. Export: envolver `canvas.toBlob` en Promise:
         ```ts
         return await new Promise<Blob>((resolve, reject) => {
           canvas.toBlob(b => b ? resolve(b) : reject(new Error('toBlob devolvió null')), 'image/jpeg', 0.85)
         })
         ```
      9. `try/catch` global: en error, `logger.error('compose failed', { err })` y re-throw para que el caller maneje.
    - NO usar `any`; tipos explícitos en todas las refs y promises.
    - NO subscribirse a nada externo — el composable es puro, no necesita `cleanup()`.
    - Verificar `cd el-templo-app && pnpm exec vue-tsc --noEmit`.
  </action>
