```ts
interface BarChallengeStore {
  // state
  startTimestamp: Ref<number | null>;
```

       ```ts
       async function submit(): Promise<void> {
         if (!attemptResult.value) {
           logger.warn('submit called without attemptResult')
           return
         }
         const payload = { secondsHeld: attemptResult.value.seconds }
         const delays = [1000, 3000, 9000]
         for (let attempt = 0; attempt <= delays.length; attempt++) {
           try {
             await api.post('/api/bar-challenge/result', payload)
             logger.info('submit success', { attempt })
             return
           } catch (err: unknown) {
             // 409 → consumed elsewhere, do not retry, do not queue
             const status = (err as { response?: { status?: number } }).response?.status
             if (status === 409) {
               logger.warn('submit returned 409 — already attempted elsewhere')
               return
             }
             if (attempt < delays.length) {
               await new Promise(r => setTimeout(r, delays[attempt]))
               continue
             }
             // All retries exhausted → enqueue
             enqueuePending(payload.secondsHeld)
             logger.error('submit failed after 3 retries — queued', { err: err instanceof Error ? err.message : String(err) })
             return
           }
         }
       }
       ```
    5. Helper functions:
       ```ts
       function enqueuePending(secondsHeld: number) {
         try {
           const raw = sessionStorage.getItem(SESSION_KEY)
           const queue: PendingSubmit[] = raw ? (JSON.parse(raw) as PendingSubmit[]) : []
           queue.push({ secondsHeld, queuedAt: Date.now() })
           sessionStorage.setItem(SESSION_KEY, JSON.stringify(queue))
         } catch (err: unknown) {
           logger.error('enqueuePending failed', { err: err instanceof Error ? err.message : String(err) })
         }
       }

       async function drainPendingSubmits(): Promise<void> {
         try {
           const raw = sessionStorage.getItem(SESSION_KEY)
           if (!raw) return
           const queue: PendingSubmit[] = JSON.parse(raw) as PendingSubmit[]
           const remaining: PendingSubmit[] = []
           for (const entry of queue) {
             try {
               await api.post('/api/bar-challenge/result', { secondsHeld: entry.secondsHeld })
               // success — drop
             } catch (err: unknown) {
               const status = (err as { response?: { status?: number } }).response?.status
               if (status === 409) {
                 // consumed — drop
                 continue
               }
               remaining.push(entry)
             }
           }
           if (remaining.length === 0) sessionStorage.removeItem(SESSION_KEY)
           else sessionStorage.setItem(SESSION_KEY, JSON.stringify(remaining))
         } catch (err: unknown) {
           logger.error('drain failed', { err: err instanceof Error ? err.message : String(err) })
         }
       }
       ```
