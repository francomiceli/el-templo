```typescript
/**
 * Level display utilities for El Templo brand identity
 *
 * Maps member level names to Greek letters:
 * - Alfa  -> alpha (lowercase Greek alpha)
 * - Delta -> Delta (uppercase Greek Delta - triangle)
 * - Sigma -> Sigma (uppercase Greek Sigma)
 * - Omega -> Omega (uppercase Greek Omega)
 *
 * Note: Alfa uses lowercase alpha for visual distinction,
 * while Delta/Sigma/Omega use uppercase for their iconic shapes.
 */

/**
 * Map of level names to Greek letter Unicode characters
 */
export const LEVEL_GREEK_MAP: Record<string, string> = {
  alfa: '\u03B1',     // α (lowercase alpha)
  delta: '\u0394',    // Δ (uppercase Delta)
  sigma: '\u03A3',    // Σ (uppercase Sigma)
  omega: '\u03A9',    // Ω (uppercase Omega)
  spartan: '\u03A9',  // Ω (same as omega - highest level)
};

/**
 * Get Greek letter for a given level
 *
 * @param level - Level name (case-insensitive: "Alfa", "alfa", "ALFA")
 * @returns Greek letter character or original level if not found
 *
 * @example
 * getLevelGreek('Alfa')   // Returns: 'α'
 * getLevelGreek('DELTA')  // Returns: 'Δ'
 * getLevelGreek('sigma')  // Returns: 'Σ'
 * getLevelGreek('Omega')  // Returns: 'Ω'
 * getLevelGreek('Unknown')// Returns: 'Unknown'
 */
export function getLevelGreek(level: string): string {
  const normalizedLevel = level.toLowerCase();
  return LEVEL_GREEK_MAP[normalizedLevel] || level;
}

/**
 * Get display text combining Greek letter with level name
 *
 * @param level - Level name (case-insensitive)
 * @returns Formatted string like "α Alfa" or original level if not mapped
 *
 * @example
 * getLevelDisplayFull('delta') // Returns: 'Δ Delta'
 */
export function getLevelDisplayFull(level: string): string {
  const greek = getLevelGreek(level);
  if (greek === level) {
    // Not mapped, return as-is
    return level;
  }
  // Capitalize first letter of level name
  const capitalizedLevel = level.charAt(0).toUpperCase() + level.slice(1).toLowerCase();
  return `${greek} ${capitalizedLevel}`;
}

/**
 * Get the proper capitalized level name
 *
 * @param level - Level name in any case
 * @returns Properly capitalized level name
 *
 * @example
 * formatLevelName('ALFA')  // Returns: 'Alfa'
 * formatLevelName('delta') // Returns: 'Delta'
 */
export function formatLevelName(level: string): string {
  return level.charAt(0).toUpperCase() + level.slice(1).toLowerCase();
}
```

Key design decisions:
- Case-insensitive lookup (normalizes to lowercase)
- Returns original input if level not found (graceful fallback)
- Spartan maps to Omega (both are highest tier)
- Alpha is lowercase for visual distinction from Delta
- Exports both map constant and function for flexibility
