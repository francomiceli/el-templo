import { ref, type Ref } from 'vue';
import { api } from 'src/boot/axios';
import { useSessionPlayerStore, type SessionEvent } from '../stores/sessionPlayerStore';
import type { BlockRole, DayBlock } from '../types/session';

export interface BlockCompletion {
  blockRole: string;
  startedAt: string;
  completedAt: string;
  exerciseCount: number;
  skipped: boolean;
}

export interface CompletionData {
  dayId: string;
  date: string;
  rpe: number | null;
  notes: string | null;
  totalDurationSeconds: number;
  blocksCompleted: BlockCompletion[];
  events: SessionEvent[];
}

export interface CompletionResponse {
  success: boolean;
  completedSessionId: number;
  totalDaysTrained: number;
}

export interface SessionCompletionReturn {
  /** Whether API call is in progress */
  isSubmitting: Ref<boolean>;
  /** Total days trained (populated after first API call) */
  totalDaysTrained: Ref<number>;
  /** Error message if submission failed */
  error: Ref<string | null>;
  /** Record a session event */
  recordEvent: (dayId: string, type: SessionEvent['type'], blockRole?: BlockRole) => Promise<void>;
  /** Get block completion data from timestamps */
  buildBlockCompletions: (
    playedBlocks: DayBlock[],
    events: SessionEvent[]
  ) => BlockCompletion[];
  /** Submit completion to API */
  completeSession: (data: CompletionData) => Promise<CompletionResponse | null>;
  /** Clear events after successful submission */
  clearEvents: (dayId: string) => Promise<void>;
}

/**
 * Composable for session completion logic
 *
 * Handles event recording, block completion building, and API submission
 */
export function useSessionCompletion(): SessionCompletionReturn {
  const store = useSessionPlayerStore();

  const isSubmitting = ref(false);
  const totalDaysTrained = ref(0);
  const error = ref<string | null>(null);

  /**
   * Record a session event (delegates to store)
   */
  async function recordEvent(
    dayId: string,
    type: SessionEvent['type'],
    blockRole?: BlockRole
  ): Promise<void> {
    await store.recordEvent(dayId, type, blockRole);
  }

  /**
   * Build BlockCompletion array from events
   */
  function buildBlockCompletions(
    playedBlocks: DayBlock[],
    events: SessionEvent[]
  ): BlockCompletion[] {
    const completions: BlockCompletion[] = [];

    for (const block of playedBlocks) {
      const startEvent = events.find(
        e => e.type === 'block_started' && e.blockRole === block.role
      );
      const completeEvent = events.find(
        e => e.type === 'block_completed' && e.blockRole === block.role
      );

      completions.push({
        blockRole: block.role,
        startedAt: startEvent?.timestamp ?? new Date().toISOString(),
        completedAt: completeEvent?.timestamp ?? new Date().toISOString(),
        exerciseCount: block.exercises?.length ?? 0,
        skipped: false,
      });
    }

    return completions;
  }

  /**
   * Submit completion data to API
   */
  async function completeSession(
    data: CompletionData
  ): Promise<CompletionResponse | null> {
    isSubmitting.value = true;
    error.value = null;

    try {
      const response = await api.post<CompletionResponse>('/sessions/complete', {
        dayId: data.dayId,
        date: data.date,
        rpe: data.rpe,
        notes: data.notes,
        totalDurationSeconds: data.totalDurationSeconds,
        blocksCompleted: data.blocksCompleted.map(b => ({
          blockRole: b.blockRole,
          startedAt: b.startedAt,
          completedAt: b.completedAt,
          skipped: b.skipped,
        })),
        events: data.events,
      });

      totalDaysTrained.value = response.data.totalDaysTrained;
      return response.data;
    } catch (err: unknown) {
      const message = err instanceof Error ? err.message : 'Failed to complete session';
      error.value = message;
      console.error('Session completion failed:', err);
      return null;
    } finally {
      isSubmitting.value = false;
    }
  }

  /**
   * Clear events after successful submission
   */
  async function clearEvents(dayId: string): Promise<void> {
    await store.clearEvents(dayId);
  }

  return {
    isSubmitting,
    totalDaysTrained,
    error,
    recordEvent,
    buildBlockCompletions,
    completeSession,
    clearEvents,
  };
}

async function onSummaryFinish(data: { rpe: number | null; notes: string | null }): Promise<void> {
  if (!session.value || !player.value) return;

  const events = await useSessionPlayerStore().getEvents(session.value.dayId);

  const result = await completeSession({

  if (result) {
