1. In `routes.ts`, add a new route:
   ```typescript
   // GET /admin/sessions/:id/pdf - Generate and download session PDF
   fastify.get<{ Params: { id: number } }>('/sessions/:id/pdf', {
     schema: {
       params: { type: 'object', properties: { id: { type: 'number' } }, required: ['id'] },
     },
   }, async (request, reply) => {
     const session = await adminService.getSessionWithDetails(request.params.id);
     if (!session) {
       return reply.status(404).send({ error: 'Sesion no encontrada' });
     }
     if (session.status !== 'approved') {
       return reply.status(400).send({ error: 'Solo sesiones aprobadas pueden generar PDF' });
     }

     const html = pdfService.buildSessionHtml(session);
     const pdfBuffer = await pdfService.generatePdf(html);

     const dayLabels: Record<number, string> = { 1: 'Lunes', 2: 'Martes', 3: 'Miercoles', 4: 'Jueves', 5: 'Viernes', 6: 'Sabado' };
     const dayName = dayLabels[session.day] || `Dia${session.day}`;
     const filename = `Sesion-S${session.week}-${dayName}-${session.levelGroup}.pdf`;

     reply.header('Content-Type', 'application/pdf');
     reply.header('Content-Disposition', `attachment; filename="${filename}"`);
     return reply.send(pdfBuffer);
   });
   ```

   ```typescript
   async function downloadSessionPdf(sessionId: number): Promise<void> {
     const response = await api.get(`/admin/sessions/${sessionId}/pdf`, {
       responseType: 'blob',
     });
     // Create download link
     const url = window.URL.createObjectURL(new Blob([response.data], { type: 'application/pdf' }));
     const link = document.createElement('a');
     link.href = url;
     // Extract filename from Content-Disposition header if available, or use default
     const disposition = response.headers['content-disposition'];
     const filenameMatch = disposition?.match(/filename="(.+)"/);
     link.download = filenameMatch?.[1] || `session-${sessionId}.pdf`;
     document.body.appendChild(link);
     link.click();
     document.body.removeChild(link);
     window.URL.revokeObjectURL(url);
   }
   ```
