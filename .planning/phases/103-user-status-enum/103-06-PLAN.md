    import { describe, it, expect, beforeEach, afterAll } from 'vitest';
    import { createTestApp, cleanAllTestData, getAuthToken, createStaffUser } from '../helpers';
    import { eq } from 'drizzle-orm';
    import * as schema from '../../src/db/schema';

    describe('Phase 103 — PATCH /api/admin/users/:id/status (R11)', () => {
      let app: Awaited<ReturnType<typeof createTestApp>>;
      let adminToken: string;
      let coachId: number;

      beforeEach(async () => {
        app = await createTestApp();
        await cleanAllTestData(app);
        adminToken = await getAuthToken(app, /* admin creds */);
        const coach = await createStaffUser(app, {
          email: 'coach1@test.com',
          role: 'coach',
          password: 'xxxxxxxx',
        });
        coachId = coach.id;
      });

      afterAll(async () => app.close());

      it('PATCH { disabled: true } sets users.staff_disabled=1', async () => {
        const res = await app.inject({
          method: 'PATCH', url: `/api/admin/users/${coachId}/status`,
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { disabled: true },
        });
        expect(res.statusCode).toBe(200);
        const body = res.json();
        expect(body).toHaveProperty('staffDisabled', true);
        expect(body).not.toHaveProperty('isActive');
        const [u] = await app.db.select().from(schema.users).where(eq(schema.users.id, coachId));
        expect(u.staffDisabled).toBe(true);
      });

      it('PATCH { disabled: false } restores active staff member', async () => {
        await app.db.update(schema.users).set({ staffDisabled: true }).where(eq(schema.users.id, coachId));
        const res = await app.inject({
          method: 'PATCH', url: `/api/admin/users/${coachId}/status`,
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { disabled: false },
        });
        expect(res.statusCode).toBe(200);
        const [u] = await app.db.select().from(schema.users).where(eq(schema.users.id, coachId));
        expect(u.staffDisabled).toBe(false);
      });

      it('rejects PATCH with old { isActive: true } payload (400)', async () => {
        const res = await app.inject({
          method: 'PATCH', url: `/api/admin/users/${coachId}/status`,
          headers: { Authorization: `Bearer ${adminToken}` },
          payload: { isActive: true },
        });
        expect(res.statusCode).toBe(400);
      });

      it('rejects PATCH without admin auth', async () => {
        const res = await app.inject({
          method: 'PATCH', url: `/api/admin/users/${coachId}/status`,
          payload: { disabled: true },
        });
        expect([401, 403]).toContain(res.statusCode);
      });
    });
