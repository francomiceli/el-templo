    5. Create `el-templo-api/test/helpers.ts`:
       ```typescript
       import { buildApp } from '../src/app';
       import type { FastifyInstance } from 'fastify';

       export async function createTestApp(): Promise<FastifyInstance> {
         process.env.DB_NAME = 'eltemplo_test';
         process.env.NODE_ENV = 'test';
         process.env.JWT_SECRET = 'test-secret-for-testing';
         const app = await buildApp();
         await app.ready();
         return app;
       }

       export async function getAuthToken(app: FastifyInstance, email: string, password: string): Promise<string> {
         const response = await app.inject({
           method: 'POST',
           url: '/api/auth/login',
           payload: { email, password },
         });
         const body = JSON.parse(response.body);
         return body.token;
       }

       export async function registerUser(app: FastifyInstance, data: {
         email: string; password: string; name: string; branchId: number; level?: string;
       }): Promise<{ token: string; user: any }> {
         const response = await app.inject({
           method: 'POST',
           url: '/api/auth/register',
           payload: data,
         });
         return JSON.parse(response.body);
       }
       ```

    Important: The existing `buildApp()` factory in app.ts already supports this pattern. No changes to app.ts are needed -- just set env vars before calling buildApp() and it will connect to the test database.

  <action>
    Create integration tests for authentication routes using Fastify inject(). Test the following scenarios:

    **POST /api/auth/register:**
    - Success: valid email/password/name/branchId returns 200 with token + user object
    - Validation: missing email returns 400
    - Validation: short password (< 6 chars) returns 400
    - Duplicate: registering same email twice returns 409 or appropriate error

    **POST /api/auth/login:**
    - Success: correct credentials return 200 with token
    - Wrong password: returns 401
    - Nonexistent email: returns 401 (same error as wrong password for security)

    **Protected route access:**
    - Request without Authorization header returns 401
    - Request with invalid/malformed token returns 401
    - Request with valid token succeeds

    Pattern:
    ```typescript
    import { describe, it, expect, beforeAll, afterAll } from 'vitest';
    import { createTestApp } from '../helpers';

    describe('Auth Routes', () => {
      let app;
      beforeAll(async () => { app = await createTestApp(); });
      afterAll(async () => { await app.close(); });

      describe('POST /api/auth/register', () => {
        it('registers a new user with valid data', async () => {
          const res = await app.inject({
            method: 'POST',
            url: '/api/auth/register',
            payload: { email: 'test@example.com', password: 'password123', name: 'Test User', branchId: 1 },
          });
          expect(res.statusCode).toBe(200);
          const body = JSON.parse(res.body);
          expect(body).toHaveProperty('token');
          expect(body.user).toHaveProperty('email', 'test@example.com');
        });
        // ... more tests
      });
    });
    ```

    Adapt the test data to match the actual schema and route signatures. Read the auth routes file to understand exact request/response shapes. Use the seed data from setup.ts (branchId from seeded branch).

    Target: 6-8 test cases covering the auth domain.
