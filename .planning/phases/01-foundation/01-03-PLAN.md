```typescript
import { mysqlTable, int, varchar, boolean, timestamp } from 'drizzle-orm/mysql-core';
import { relations } from 'drizzle-orm';

export const branches = mysqlTable('branches', {
  id: int('id').primaryKey().autoincrement(),
  name: varchar('name', { length: 255 }).notNull(),
  code: varchar('code', { length: 10 }).notNull().unique(),
  isActive: boolean('is_active').default(true).notNull(),
  createdAt: timestamp('created_at').defaultNow().notNull(),
  updatedAt: timestamp('updated_at').defaultNow().onUpdateNow().notNull(),
});

// Relations defined after users import to avoid circular dependency
export const branchesRelations = relations(branches, ({ many }) => ({
  users: many(users),
}));

// Import at bottom to avoid circular reference
import { users } from './users';
```

2. Create src/db/schema/users.ts:
```typescript
import { mysqlTable, int, varchar, timestamp, mysqlEnum } from 'drizzle-orm/mysql-core';
import { relations } from 'drizzle-orm';
import { branches } from './branches';

export const roleEnum = mysqlEnum('role', ['member', 'coach', 'admin', 'superadmin']);
export const levelEnum = mysqlEnum('level', ['alfa', 'delta', 'sigma', 'omega', 'spartan']);

export const users = mysqlTable('users', {
  id: int('id').primaryKey().autoincrement(),
  email: varchar('email', { length: 255 }).notNull().unique(),
  passwordHash: varchar('password_hash', { length: 255 }).notNull(),
  firstName: varchar('first_name', { length: 100 }),
  lastName: varchar('last_name', { length: 100 }),
  role: roleEnum.default('member').notNull(),
  branchId: int('branch_id').references(() => branches.id).notNull(),
  level: levelEnum.default('alfa').notNull(),
  createdAt: timestamp('created_at').defaultNow().notNull(),
  updatedAt: timestamp('updated_at').defaultNow().onUpdateNow().notNull(),
});

export const usersRelations = relations(users, ({ one }) => ({
  branch: one(branches, {
    fields: [users.branchId],
    references: [branches.id],
  }),
}));
```

```typescript
import 'dotenv/config';
import { drizzle } from 'drizzle-orm/mysql2';
import mysql from 'mysql2/promise';
import * as argon2 from 'argon2';
import { branches, users } from './schema';

// Real El Templo branch locations (per CONTEXT.md)
const BRANCHES = [
  { name: 'Centro', code: 'CENTRO' },
  { name: 'Alem', code: 'ALEM' },
  { name: 'Constitucion', code: 'CONST' },
  { name: 'Jujuy', code: 'JUJUY' },
  { name: 'Mogotes', code: 'MOGOT' },
];

// Level distribution for test users
const LEVELS = ['alfa', 'delta', 'sigma', 'omega', 'spartan'] as const;

async function seed() {
  console.log('Connecting to database...');

  const connection = await mysql.createConnection({
    host: process.env.DB_HOST || 'localhost',
    port: Number(process.env.DB_PORT) || 3306,
    user: process.env.DB_USER || 'root',
    password: process.env.DB_PASSWORD || '',
    database: process.env.DB_NAME || 'eltemplo',
  });

  const db = drizzle(connection, { mode: 'default' });

  try {
    // Clear existing data (order matters due to foreign keys)
    console.log('Clearing existing data...');
    await db.delete(users);
    await db.delete(branches);

    // Seed branches
    console.log('Seeding branches...');
    const insertedBranches = await db.insert(branches).values(BRANCHES).$returningId();
    console.log(`  Seeded ${insertedBranches.length} branches`);

    // Create branch ID lookup
    const branchIds = insertedBranches.map(b => b.id);

    // Hash password once for all test users (same password for dev convenience)
    const testPasswordHash = await argon2.hash('password123', {
      type: argon2.argon2id,
      memoryCost: 19456,
      timeCost: 2,
      parallelism: 1,
    });

    // Seed superadmin (per CONTEXT.md: admin@eltemplo.com)
    console.log('Seeding superadmin...');
    const adminPasswordHash = await argon2.hash('admin123', {
      type: argon2.argon2id,
      memoryCost: 19456,
      timeCost: 2,
      parallelism: 1,
    });

    await db.insert(users).values({
      email: 'admin@eltemplo.com',
      passwordHash: adminPasswordHash,
      firstName: 'Super',
      lastName: 'Admin',
      role: 'superadmin',
      branchId: branchIds[0], // Centro
      level: 'spartan',
    });
    console.log('  Seeded superadmin: admin@eltemplo.com / admin123');

    // Seed test users: 1 coach + 4 members per branch
    console.log('Seeding test users...');
    let userCount = 0;

    for (let i = 0; i < BRANCHES.length; i++) {
      const branchId = branchIds[i];
      const branchCode = BRANCHES[i].code.toLowerCase();

      // 1 coach per branch
      await db.insert(users).values({
        email: `coach.${branchCode}@eltemplo.com`,
        passwordHash: testPasswordHash,
        firstName: 'Coach',
        lastName: BRANCHES[i].name,
        role: 'coach',
        branchId,
        level: 'omega', // Coaches are experienced
      });
      userCount++;

      // 4 members per branch, different levels
      for (let j = 0; j < 4; j++) {
        await db.insert(users).values({
          email: `member${j + 1}.${branchCode}@eltemplo.com`,
          passwordHash: testPasswordHash,
          firstName: `Member${j + 1}`,
          lastName: BRANCHES[i].name,
          role: 'member',
          branchId,
          level: LEVELS[j % LEVELS.length], // Distribute levels
        });
        userCount++;
      }
    }
    console.log(`  Seeded ${userCount} test users (5 coaches + 20 members)`);

    console.log('\nSeed complete!');
    console.log('\nTest credentials:');
    console.log('  Superadmin: admin@eltemplo.com / admin123');
    console.log('  Coach:      coach.centro@eltemplo.com / password123');
    console.log('  Member:     member1.centro@eltemplo.com / password123');

  } finally {
    await connection.end();
  }
}

seed().catch((err) => {
  console.error('Seed failed:', err);
  process.exit(1);
});
```
