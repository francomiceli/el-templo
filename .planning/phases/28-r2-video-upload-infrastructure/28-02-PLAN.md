**1. Create useVideoUpload composable** (`src/composables/useVideoUpload.ts`):
Manages the presigned URL upload flow. Uses `api` from `src/boot/axios` and raw `axios` for the R2 PUT.

```typescript
export function useVideoUpload() {
  const uploading = ref<Map<number, number>>(new Map()); // exerciseId -> progress %

  async function uploadVideo(
    exerciseId: number,
    file: File,
    onComplete?: () => void,
  ): Promise<void> {
    // Client-side validation
    if (file.size > 20 * 1024 * 1024) {
      Notify.create({
        type: "negative",
        message: `Archivo demasiado grande (max 20MB)`,
      });
      return;
    }
    if (!file.type.includes("mp4") && !file.name.endsWith(".mp4")) {
      Notify.create({
        type: "negative",
        message: "Solo se aceptan archivos MP4",
      });
      return;
    }

    // Client-side duration validation using HTML5 video element
    const duration = await getVideoDuration(file);
    if (duration > 20) {
      Notify.create({
        type: "negative",
        message: `Video demasiado largo (${Math.round(duration)}s, max 20s)`,
      });
      return;
    }

    uploading.value.set(exerciseId, 0);
    try {
      // Step 1: Get presigned URL from API
      const { data } = await api.post(
        `/admin/exercises/${exerciseId}/upload-url`,
      );

      // Step 2: Upload directly to R2 (no auth header)
      await axios.put(data.uploadUrl, file, {
        headers: { "Content-Type": "video/mp4" },
        onUploadProgress: (e) => {
          if (e.total) {
            uploading.value.set(
              exerciseId,
              Math.round((e.loaded / e.total) * 100),
            );
          }
        },
      });

      // Step 3: Confirm upload
      await api.post(`/admin/exercises/${exerciseId}/upload-complete`, {
        key: data.key,
      });

      Notify.create({
        type: "positive",
        message: "Video subido correctamente",
      });
      onComplete?.();
    } catch (err) {
      log.error("Upload failed", err);
      Notify.create({ type: "negative", message: "Error al subir el video" });
    } finally {
      uploading.value.delete(exerciseId);
    }
  }

  function isUploading(exerciseId: number): boolean {
    return uploading.value.has(exerciseId);
  }

  function getProgress(exerciseId: number): number {
    return uploading.value.get(exerciseId) ?? 0;
  }

  return { uploadVideo, isUploading, getProgress, uploading };
}
```

Helper function `getVideoDuration(file: File): Promise<number>`:
Create a temporary URL with `URL.createObjectURL(file)`, create a `<video>` element, set src, listen for `loadedmetadata` event, return `video.duration`. Revoke the object URL after.
