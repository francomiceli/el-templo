```typescript
export interface KpiStats {
  activeMembers: number;
  monthlyRevenue: number;
  dailyAttendanceAvg: number;
  morososCount: number;
  // Each with trend sub-object:
  activeMembersTrend: { direction: "up" | "down" | "flat"; percentage: number };
  monthlyRevenueTrend: {
    direction: "up" | "down" | "flat";
    percentage: number;
  };
  dailyAttendanceAvgTrend: {
    direction: "up" | "down" | "flat";
    percentage: number;
  };
  morososCountTrend: { direction: "up" | "down" | "flat"; percentage: number };
}

export interface AttentionMember {
  userId: number;
  firstName: string | null;
  lastName: string | null;
  planName: string;
  phone: string | null;
  type: "expiring" | "overdue";
  daysUntilExpiry: number | null;
  daysOverdue: number | null;
}

export interface MemberAnalytics {
  newMembers: number;
  churnedMembers: number;
  retentionRate: number;
  planDistribution: Array<{ planName: string; count: number }>;
  attentionList: AttentionMember[];
}

export interface HeatmapCell {
  dayOfWeek: number; // 1-6 Mon-Sat
  hour: number; // 6-22
  averageOccupancy: number; // 0-100
}

export interface SlotOccupancy {
  scheduleId: number;
  activityName: string;
  dayOfWeek: number;
  startTime: string;
  averageOccupancy: number;
}

export interface AttendanceAnalytics {
  dailyCheckins: Array<{ date: string; count: number }>;
  peakHoursHeatmap: HeatmapCell[];
  slotOccupancy: SlotOccupancy[];
  noShowRate: number;
}

export interface FinancialAnalytics {
  revenueTrend: Array<{ month: string; revenue: number }>;
  revenueByMethod: { cash: number; transfer: number; card: number };
  revenueByBranch: Array<{
    branchId: number;
    branchName: string;
    revenue: number;
  }>;
  totalOutstanding: number;
  collectionRate: number;
}
```
