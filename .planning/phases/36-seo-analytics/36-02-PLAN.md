    Create a server route at `el-templo-web/server/api/__sitemap__/blog.ts` that fetches published blog post slugs from the API at build time and returns them as sitemap entries:
    ```typescript
    import { defineSitemapEventHandler } from '#imports';

    export default defineSitemapEventHandler(async () => {
      const config = useRuntimeConfig();
      try {
        const response = await $fetch<{ posts: Array<{ slug: string; updatedAt: string }> }>(
          `${config.public.apiUrl}/api/blog/posts?limit=1000`
        );
        return response.posts.map(post => ({
          loc: `/blog/${post.slug}`,
          lastmod: post.updatedAt,
        }));
      } catch {
        return [];
      }
    });
    ```

    **Important:** The `@nuxtjs/sitemap` module auto-discovers static routes from the pages directory. The server route adds dynamic blog post URLs on top.
