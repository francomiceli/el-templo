**3. Create tag browse page** `pages/blog/tag/[slug].vue`:

This page is structurally similar to the blog index but filtered by tag.

```vue
<script setup lang="ts">
interface BlogPost { /* same as blog index */ }
interface TagWithCount { id: number; name: string; slug: string; postCount: number; }
interface TagPostsResponse {
  posts: BlogPost[];
  total: number;
  page: number;
  totalPages: number;
  tag: { id: number; name: string; slug: string };
}

const route = useRoute();
const config = useRuntimeConfig();
const tagSlug = computed(() => String(route.params.slug));
const page = computed(() => Number(route.query.page) || 1);

// Fetch tag posts
const { data, status } = useFetch<TagPostsResponse>(
  () => `${config.public.apiUrl}/blog/tags/${tagSlug.value}/posts?page=${page.value}&limit=10`,
  { watch: [page, tagSlug] },
);

// Fetch all tags for the tag bar
const { data: tagsData } = useFetch<{ tags: TagWithCount[] }>(
  () => `${config.public.apiUrl}/blog/tags`,
);

const posts = computed(() => data.value?.posts ?? []);
const tag = computed(() => data.value?.tag);
const totalPages = computed(() => data.value?.totalPages ?? 0);
const tags = computed(() => tagsData.value?.tags ?? []);
const isLoading = computed(() => status.value === 'pending');
const isEmpty = computed(() => status.value === 'success' && posts.value.length === 0);

function handlePageChange(newPage: number): void {
  navigateTo({
    path: `/blog/tag/${tagSlug.value}`,
    query: newPage > 1 ? { page: newPage } : {},
  });
}
```
