```typescript
// Documented API from drizzle-orm/utils.d.ts:37
import { getTableColumns, Table } from "drizzle-orm";
import * as schema from "../../src/db/schema/index";

// Build the rename map by walking every exported table and reading
// each column's runtime `.name` (the SQL column name; Drizzle stores
// the first arg to `mysqlEnum("subscription_status", [...])` here).
function buildRenameMap(): Map<string, Map<string, string>> {
  const map = new Map<string, Map<string, string>>(); // tableName -> (jsProperty -> sqlColumn)
  for (const [exportName, exportValue] of Object.entries(schema)) {
    // Skip non-table exports (enums, relations, etc.)
    if (!(exportValue instanceof Table)) continue;
    const cols = getTableColumns(exportValue);
    const tableName =
      (exportValue as unknown as { [Symbol.for("drizzle:Name")]?: string })[
        Symbol.for("drizzle:Name")
      ] ?? exportName;
    const colMap = new Map<string, string>();
    for (const [jsProperty, col] of Object.entries(cols)) {
      const sqlColumn = (col as { name: string }).name;
      if (jsProperty !== sqlColumn) {
        colMap.set(jsProperty, sqlColumn);
      }
    }
    if (colMap.size > 0) {
      map.set(tableName, colMap);
    }
  }
  return map;
}
```

```typescript
// Stateful accumulator: when we see `` sql` `` open, capture lines until
// the closing backtick on a subsequent line. Returns array of
// { file, line: number, sql: string } entries.
function findRawSqlSites(
  files: string[],
): Array<{ file: string; line: number; sql: string }> {
  const sites: Array<{ file: string; line: number; sql: string }> = [];
  for (const file of files) {
    const content = fs.readFileSync(file, "utf-8");
    const lines = content.split("\n");
    let inBlock = false;
    let blockLines: string[] = [];
    let startLine = 0;
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      if (!inBlock) {
        const openIdx = line.search(/\bsql`/);
        if (openIdx !== -1) {
          inBlock = true;
          startLine = i + 1; // 1-indexed
          blockLines = [line.slice(openIdx + 4)];
          // Single-line template
          const closeIdx = blockLines[0].indexOf("`");
          if (closeIdx !== -1) {
            sites.push({
              file,
              line: startLine,
              sql: blockLines[0].slice(0, closeIdx),
            });
            inBlock = false;
            blockLines = [];
          }
        }
      } else {
        const closeIdx = line.indexOf("`");
        if (closeIdx !== -1) {
          blockLines.push(line.slice(0, closeIdx));
          sites.push({ file, line: startLine, sql: blockLines.join("\n") });
          inBlock = false;
          blockLines = [];
        } else {
          blockLines.push(line);
        }
      }
    }
  }
  return sites;
}
```

**D-02 (i) — parse `FROM <table> [<alias>]` and `JOIN <table> <alias>` to know which tables are in scope:**

```typescript
function parseTablesInScope(
  sql: string,
): Array<{ table: string; alias: string }> {
  const tables: Array<{ table: string; alias: string }> = [];
  // FROM <table> [<alias>]
  const fromRe = /\bFROM\s+([a-z_][a-z0-9_]*)\s*(?:AS\s+)?([a-z][a-z0-9_]*)?/gi;
  // JOIN <table> <alias> (alias usually required for joins)
  const joinRe = /\bJOIN\s+([a-z_][a-z0-9_]*)\s*(?:AS\s+)?([a-z][a-z0-9_]*)?/gi;
  for (const re of [fromRe, joinRe]) {
    let m: RegExpExecArray | null;
    while ((m = re.exec(sql)) !== null) {
      tables.push({ table: m[1], alias: m[2] ?? m[1] });
    }
  }
  return tables;
}
```
