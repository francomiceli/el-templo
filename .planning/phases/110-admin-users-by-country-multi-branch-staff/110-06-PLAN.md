**2. Rewrite GET /admin/members/branches** (lines 119-130) to filter by `request.scope` per D-07/D-08/D-09. Replace the entire handler with:

```typescript
// GET /admin/members/branches — list active branches, filtered by request.scope.
// D-07: owner sees country-toggle filtered (or all when no ?country=); admin/gestion
// sees own country; coach/recepción sees scope.branchIds. D-09: virtual sedes
// (Templo Online) always included. Response shape preserved: { branches: [{ id, name }] }.
fastify.get("/branches", async (request) => {
  const allRows = await fastify.db
    .select({
      id: schema.branches.id,
      name: schema.branches.name,
      country: schema.branches.country,
      isVirtual: schema.branches.isVirtual,
    })
    .from(schema.branches)
    .where(eq(schema.branches.isActive, true))
    .orderBy(schema.branches.name);

  const { isOwner, country, branchIds, role } = request.scope;
  let filtered = allRows;

  if (isOwner) {
    // D-08: owner with ?country= filters; without ?country= sees all (virtual + real).
    // attachCountryScope sets country='AR' (default) or 'ES' from the toggle.
    // To support "owner sees all when no toggle", check the raw query param —
    // if absent, do NOT filter by country.
    const q = (request.query as Record<string, unknown> | undefined)?.country;
    if (q === "AR" || q === "ES") {
      filtered = allRows.filter((b) => b.isVirtual || b.country === country);
    }
    // else: owner without ?country= → keep allRows
  } else if (role === "admin" || role === "gestion") {
    // When scope.country=null (data corruption fail-closed), filter degenerates
    // to virtual-only (default-deny lateral) — consistent with canAccessBranch
    // Rule 3 evaluating false for all real branches.
    filtered = allRows.filter((b) => b.isVirtual || b.country === country);
  } else if (role === "coach" || role === "recepcion") {
    const allowed = new Set(branchIds);
    filtered = allRows.filter((b) => b.isVirtual || allowed.has(b.id));
  }
  // member: leave allRows (member-app does not call this admin endpoint;
  // module guard at top of file restricts to ALL_STAFF_ROLES anyway).

  return {
    branches: filtered.map(({ id, name }) => ({ id, name })),
  };
});
```
