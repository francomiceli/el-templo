Create a `deploy/nginx/` directory with 3 server block config files. These are HTTP-only initial configs — certbot's `--nginx` plugin will add SSL directives and HTTP→HTTPS redirects automatically during manual setup (Plan 03).

**IMPORTANT:** Write these as HTTP-only (port 80) configs. Certbot will modify them in-place to add SSL. Do NOT include SSL directives — that creates a chicken-and-egg problem where Nginx won't start without the cert files that don't exist yet.

1. **deploy/nginx/app.eltemplo.org** — Member SPA (hash mode, but include try_files for defense-in-depth):

```nginx
# Member App (SPA - hash mode routing)
# Location: /etc/nginx/sites-available/app.eltemplo.org
# SSL will be added by certbot --nginx

server {
    listen 80;
    listen [::]:80;
    server_name app.eltemplo.org;

    root /var/www/member-app;
    index index.html;

    # SPA fallback (defense-in-depth for hash mode)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets aggressively (Vite adds content hashes)
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
}
```

2. **deploy/nginx/admin.eltemplo.org** — Admin SPA (history mode, REQUIRES try_files):

```nginx
# Admin App (SPA - history mode routing)
# Location: /etc/nginx/sites-available/admin.eltemplo.org
# SSL will be added by certbot --nginx
# CRITICAL: try_files is REQUIRED for history mode - without it, direct URL access returns 404

server {
    listen 80;
    listen [::]:80;
    server_name admin.eltemplo.org;

    root /var/www/admin-app;
    index index.html;

    # SPA fallback (REQUIRED for history mode routing)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Cache static assets aggressively (Vite adds content hashes)
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
}
```

3. **deploy/nginx/api.eltemplo.org** — API proxy to Fastify:

```nginx
# API Reverse Proxy (Fastify on port 3000)
# Location: /etc/nginx/sites-available/api.eltemplo.org
# SSL will be added by certbot --nginx

server {
    listen 80;
    listen [::]:80;
    server_name api.eltemplo.org;

    # Proxy ALL requests to Fastify (routes are /api/*, /health)
    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;

        # WebSocket support
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        # Pass original client info
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Timeouts
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;

        # Don't buffer responses
        proxy_buffering off;
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
}
```

Key differences from the old single deploy/nginx.conf:
- The old config uses `server_name _` (catch-all) — new configs use specific subdomain names
- The old config only proxies `/api` and `/health` — the api subdomain proxies ALL paths (Fastify handles its own routing)
- The old config has no static file serving — app/admin configs serve static files from /var/www/
- No SSL directives included — certbot adds them during manual server setup
