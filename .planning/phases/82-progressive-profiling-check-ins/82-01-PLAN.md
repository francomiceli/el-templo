    **6. Create routes** (`el-templo-api/src/modules/check-ins/routes.ts`):
    ```typescript
    export const checkInRoutes: FastifyPluginAsync = async (fastify) => {
      const service = new CheckInService(fastify.db);

      // POST / — submit a check-in answer
      fastify.post('/', {
        onRequest: [fastify.authenticate],
        schema: { body: submitCheckInBodySchema, response: { 201: checkInResponseSchema, 400: errorResponseSchema, 403: errorResponseSchema, 409: errorResponseSchema } }
      }, async (request, reply) => {
        const { userId } = request.user;
        const { questionType, value, bodyArea } = request.body as CheckInAnswer;
        try {
          await service.submitAnswer(userId, { questionType, value, bodyArea });
          return reply.status(201).send({ success: true });
        } catch (err: unknown) {
          if (err instanceof Error) {
            if (err.message === 'Ya respondiste esta pregunta hoy') return reply.status(409).send({ error: err.message });
            if (err.message === 'Pregunta no disponible todavia') return reply.status(403).send({ error: err.message });
            return reply.status(400).send({ error: err.message });
          }
          return reply.status(400).send({ error: 'Error al registrar respuesta' });
        }
      });

      // GET /today — get today's check-in state
      fastify.get('/today', {
        onRequest: [fastify.authenticate],
        schema: { response: { 200: todayCheckInResponseSchema } }
      }, async (request) => {
        const { userId } = request.user;
        return service.getTodayState(userId);
      });
    };
    ```
