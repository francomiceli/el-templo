**File 1: `el-templo-api/src/db/schema/micro-programs.ts`**

Module comment: `// Module: programs`

Table `micro_programs`:

- `id`: int, primary key, autoincrement
- `name`: varchar(150), notNull
- `description`: text, nullable
- `price`: int, notNull (in whole currency units, same pattern as priceRegular on subscription_plans)
- `durationWeeks`: int("duration_weeks"), notNull (4-8 range enforced at service layer)
- `sessionsPerWeekToAdvance`: int("sessions_per_week_to_advance"), notNull, default 3 (per D-04: configurable sessions needed to unlock next week)
- `auraWeeklyBonus`: int("aura_weekly_bonus"), default 15 (per D-31)
- `auraCompletionBonus`: int("aura_completion_bonus"), default 100 (per D-32)
- `isActive`: boolean("is_active"), default true, notNull (per D-43: deactivated stops new enrollments)
- `createdAt`: timestamp("created_at"), defaultNow, notNull
- `updatedAt`: timestamp("updated_at"), defaultNow, onUpdateNow, notNull

Table `micro_program_content_blocks`:

- `id`: int, primary key, autoincrement
- `programId`: int("program_id"), references micro_programs.id, notNull
- `weekNumber`: int("week_number"), notNull (1-based, which week this content belongs to)
- `sortOrder`: int("sort_order"), notNull, default 0 (per D-37: reorder via drag/arrows)
- `blockType`: mysqlEnum("block_type", ["video", "text", "pdf", "exercise"]) (per D-03)
- `title`: varchar("title", { length: 200 }), notNull
- `content`: text("content"), nullable (markdown text for type "text", URL for "pdf", null for "exercise")
- `videoUrl`: varchar("video_url", { length: 500 }), nullable (R2 URL for type "video", per D-03: NOT YouTube)
- `exerciseId`: int("exercise_id"), nullable, references exercises.id (for type "exercise", per D-03)
- `createdAt`: timestamp("created_at"), defaultNow, notNull

Add relations:

- `microProgramsRelations`: one-to-many to content_blocks
- `microProgramContentBlocksRelations`: many-to-one back to micro_programs, optional one to exercises

Add indexes on content_blocks:

- `index("idx_content_blocks_program_week").on(table.programId, table.weekNumber)`

**File 2: `el-templo-api/src/db/schema/program-enrollments.ts`**

Enum `programEnrollmentStatusEnum`: mysqlEnum("program_enrollment_status", ["active", "completed", "expired", "cancelled"]) (per D-28)

Table `program_enrollments`:

- `id`: int, primary key, autoincrement
- `userId`: int("user_id"), references users.id, notNull

- `status`: programEnrollmentStatusEnum, default "active", notNull

- `weekUnlockedAt`: timestamp("week_unlocked_at"), nullable (when current week was unlocked, for time gating)
- `enrolledAt`: timestamp("enrolled_at"), defaultNow, notNull (per D-27: Week 1 starts from this date)
- `completedAt`: timestamp("completed_at"), nullable
- `expiredAt`: timestamp("expired_at"), nullable
- `cancelledAt`: timestamp("cancelled_at"), nullable
- `paymentAmount`: int("payment_amount"), nullable (what they paid for this enrollment)
- `paymentMethod`: varchar("payment_method", { length: 30 }), nullable (transfer, cash, etc.)
- `notes`: text("notes"), nullable
- `createdAt`: timestamp("created_at"), defaultNow, notNull
- `updatedAt`: timestamp("updated_at"), defaultNow, onUpdateNow, notNull

Add relations:

- `programEnrollmentsRelations`: many-to-one to users, many-to-one to micro_programs

Add indexes:

- `index("idx_enrollments_user_id").on(table.userId)`
- `index("idx_enrollments_program_id").on(table.programId)`
- `index("idx_enrollments_status").on(table.status)`

// Module: programs

// ---- Content block types ----
export type ContentBlockType = "video" | "text" | "pdf" | "exercise";

export interface ContentBlockInput {
  weekNumber: number;
  sortOrder: number;
  blockType: ContentBlockType;
  title: string;
  content: string | null; // markdown for text, URL for pdf
  videoUrl: string | null; // R2 URL for video
  exerciseId: number | null; // exercise table FK for exercise type
}

export interface ContentBlockDetail {
  id: number;
  weekNumber: number;
  sortOrder: number;
  blockType: ContentBlockType;
  title: string;
  content: string | null;
  videoUrl: string | null;
  exerciseId: number | null;
  exerciseName: string | null; // joined from exercises table
  exerciseVideoUrl: string | null; // R2 video from exercises table
}

// ---- Program types ----
export interface CreateProgramInput {
  name: string;
  description: string | null;
  price: number;
  durationWeeks: number;
  sessionsPerWeekToAdvance: number;
  auraWeeklyBonus: number;
  auraCompletionBonus: number;
  contentBlocks: ContentBlockInput[];
}

export interface UpdateProgramInput {
  name?: string;
  description?: string | null;
  price?: number;
  // durationWeeks NOT editable when active enrollments exist (per D-41)
  auraWeeklyBonus?: number;
  auraCompletionBonus?: number;
}

export interface MicroProgram {
  id: number;
  name: string;
  description: string | null;
  price: number;
  durationWeeks: number;
  sessionsPerWeekToAdvance: number;
  auraWeeklyBonus: number;
  auraCompletionBonus: number;
  isActive: boolean;
  createdAt: string;
}

export interface MicroProgramDetail extends MicroProgram {
  contentBlocks: ContentBlockDetail[];
  activeEnrollmentCount: number;
}

// ---- Enrollment types ----
export type EnrollmentStatus = "active" | "completed" | "expired" | "cancelled";

export interface EnrollMemberInput {
  userId: number;
  programId: number;
  paymentAmount: number | null;
  paymentMethod: string | null;
  notes: string | null;
}

export interface ProgramEnrollment {
  id: number;
  userId: number;
  programId: number;
  programName: string;
  status: EnrollmentStatus;
  currentWeek: number;
  sessionsCompletedThisWeek: number;
  durationWeeks: number; // from joined program
  sessionsPerWeekToAdvance: number; // from joined program
  enrolledAt: string;
  completedAt: string | null;
  expiredAt: string | null;
  cancelledAt: string | null;
  paymentAmount: number | null;
  paymentMethod: string | null;
}

// ---- Member-facing types ----
export interface MemberProgramCatalogItem {
  id: number;
  name: string;
  description: string | null;
  price: number;
  durationWeeks: number;
  hasContent: boolean; // false = show "Proximamente" per D-46
}

export interface MemberEnrollmentProgress {
  enrollmentId: number;
  programName: string;
  currentWeek: number;
  durationWeeks: number;
  sessionsCompletedThisWeek: number;
  sessionsPerWeekToAdvance: number;
  isWeekComplete: boolean; // derived: sessionsCompleted >= sessionsRequired
  daysUntilExpiry: number | null; // for renewal badge per D-16
  contentBlocks: ContentBlockDetail[]; // current week's blocks only
}

// ---- Analytics types ----
export interface ProgramAnalytics {
  totalEnrollments: number;
  activeEnrollments: number;
  completedCount: number;
}
