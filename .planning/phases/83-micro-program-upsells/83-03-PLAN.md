export type ContentBlockType = "video" | "text" | "pdf" | "exercise";

**Create `el-templo-admin/src/composables/useProgramsApi.ts`:**

Follow pattern from useSubscriptionsApi.ts. Use `api` from `src/boot/axios` and `createLogger` from `src/utils/logger`.

```typescript
export function useProgramsApi() {
  // Program CRUD
  async function getPrograms(): Promise<MicroProgram[]>;
  async function getProgramDetail(
    programId: number,
  ): Promise<MicroProgramDetail>;
  async function createProgram(input: CreateProgramPayload): Promise<number>; // returns id
  async function updateProgram(
    programId: number,
    input: Partial<MicroProgram>,
  ): Promise<void>;
  async function addContentBlocks(
    programId: number,
    blocks: ContentBlockInput[],
  ): Promise<void>;
  async function deactivateProgram(programId: number): Promise<void>;

  // Enrollment management
  async function enrollMember(input: {
    userId: number;
    programId: number;
    paymentAmount: number | null;
    paymentMethod: string | null;
    notes: string | null;
  }): Promise<number>;
  async function cancelEnrollment(enrollmentId: number): Promise<void>;
  async function advanceWeek(enrollmentId: number): Promise<void>;
  async function getUserEnrollments(
    userId: number,
  ): Promise<ProgramEnrollment[]>;

  // Analytics
  async function getAnalytics(): Promise<ProgramAnalytics>;

  return {
    getPrograms,
    getProgramDetail,
    createProgram,
    updateProgram,
    addContentBlocks,
    deactivateProgram,
    enrollMember,
    cancelEnrollment,
    advanceWeek,
    getUserEnrollments,
    getAnalytics,
  };
}
```
