```typescript
import pino from 'pino';

// Base session logger with common configuration
export const sessionLogger = pino({
  level: process.env.LOG_LEVEL || 'info',
  formatters: {
    level: (label) => ({ level: label.toUpperCase() }),
  },
  timestamp: pino.stdTimeFunctions.isoTime,
  // Redact sensitive fields if any
  redact: [],
  // Use pino-pretty in development
  transport: process.env.NODE_ENV === 'development'
    ? { target: 'pino-pretty', options: { colorize: true } }
    : undefined,
});

// Create child logger with block context
export function createBlockLogger(where: TraceWhere) {
  return sessionLogger.child({
    weekId: where.weekId,
    dayId: where.dayId,
    levelGroup: where.levelGroup,
    blockId: where.blockId,
    blockRole: where.blockRole,
  });
}

// Create child logger with session context
export function createSessionLogger(weekId: number, dayId: string, levelGroup: string) {
  return sessionLogger.child({
    weekId,
    dayId,
    levelGroup,
    component: 'session-generator',
  });
}
```

**trace/emitter.ts:**
```typescript
import type { Logger } from 'pino';
import type { TraceEvent, TraceSeverity, TraceCode, TraceWhere, BlockTrace, SessionTrace } from './types';

// Create a trace event
export function createTraceEvent(
  code: TraceCode,
  where: TraceWhere,
  options: {
    severity?: TraceSeverity;
    decision?: Record<string, unknown>;
    reason?: TraceEvent['reason'];
    metrics?: Record<string, number>;
    message?: string;
  } = {}
): TraceEvent {
  return {
    ts: new Date().toISOString(),
    severity: options.severity || 'INFO',
    code,
    where,
    decision: options.decision,
    reason: options.reason,
    metrics: options.metrics,
    message: options.message,
  };
}

// Emit trace to logger AND return event for collection
export function emitTrace(
  logger: Logger,
  event: TraceEvent
): TraceEvent {
  const logData = {
    trace: event,
    code: event.code,
    where: event.where,
  };

  switch (event.severity) {
    case 'INFO':
      logger.info(logData, event.message || event.code);
      break;
    case 'WARNING':
      logger.warn(logData, event.message || event.code);
      break;
    case 'ERROR':
    case 'HARD_ERROR':
      logger.error(logData, event.message || event.code);
      break;
  }

  return event;
}

// Aggregate block events into BlockTrace
export function aggregateBlockTrace(blockId: string, events: TraceEvent[]): BlockTrace {
  return {
    blockId,
    events,
    summary: {
      totalEvents: events.length,
      warnings: events.filter(e => e.severity === 'WARNING').length,
      errors: events.filter(e => e.severity === 'ERROR' || e.severity === 'HARD_ERROR').length,
      fallbacksApplied: events.filter(e => e.code.includes('FALLBACK')).length,
    },
  };
}

// Aggregate session traces
export function aggregateSessionTrace(
  sessionId: string,
  blockTraces: BlockTrace[],
  durationMs: number
): SessionTrace {
  return {
    sessionId,
    blockTraces,
    summary: {
      totalEvents: blockTraces.reduce((sum, b) => sum + b.summary.totalEvents, 0),
      totalWarnings: blockTraces.reduce((sum, b) => sum + b.summary.warnings, 0),
      totalErrors: blockTraces.reduce((sum, b) => sum + b.summary.errors, 0),
      generationDurationMs: durationMs,
    },
  };
}
```
