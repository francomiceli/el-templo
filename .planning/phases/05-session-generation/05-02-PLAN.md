```typescript
// sessions table - one row per generated day session
export const sessions = mysqlTable('sessions', {
  id: int('id').primaryKey().autoincrement(),
  dayId: varchar('day_id', { length: 50 }).notNull().unique(), // W1-lunes-sigma
  week: int('week').notNull(),
  day: varchar('day', { length: 20 }).notNull(), // lunes, martes, etc
  levelGroup: varchar('level_group', { length: 20 }).notNull(), // alfa_delta, sigma, omega
  blockCount: int('block_count').notNull(),
  traceJson: json('trace_json'), // Full trace for debugging
  createdAt: timestamp('created_at').defaultNow(),
}, (table) => [
  index('sessions_week_day_level_idx').on(table.week, table.day, table.levelGroup),
]);
```

**session-blocks.ts:**
```typescript
// session_blocks table - one row per block in a session
export const sessionBlocks = mysqlTable('session_blocks', {
  id: int('id').primaryKey().autoincrement(),
  sessionId: int('session_id').notNull().references(() => sessions.id, { onDelete: 'cascade' }),
  blockId: varchar('block_id', { length: 100 }).notNull(),
  role: varchar('role', { length: 20 }).notNull(), // INITIUM, NUCLEUS, etc
  route: varchar('route', { length: 20 }).notNull(),
  pattern: varchar('pattern', { length: 150 }).notNull(),
  intensity: int('intensity').notNull(),
  repsBudget: int('reps_budget').notNull(),
  formatId: int('format_id').notNull(),
  formatName: varchar('format_name', { length: 50 }).notNull(),
  exerciseCount: int('exercise_count').notNull(),
  sortOrder: int('sort_order').notNull(), // 0-4 for block ordering
}, (table) => [
  index('session_blocks_session_idx').on(table.sessionId),
]);
```

**session-prescriptions.ts:**
```typescript
// session_prescriptions - one row per exercise in a block
export const sessionPrescriptions = mysqlTable('session_prescriptions', {
  id: int('id').primaryKey().autoincrement(),
  blockId: int('block_id').notNull().references(() => sessionBlocks.id, { onDelete: 'cascade' }),
  exerciseId: int('exercise_id').notNull(),
  exerciseName: varchar('exercise_name', { length: 150 }).notNull(),
  contraction: varchar('contraction', { length: 10 }).notNull(), // CON, EXC, ISO
  reps: int('reps').notNull(),
  seconds: int('seconds').notNull(),
  rest: int('rest').notNull(),
  notes: varchar('notes', { length: 255 }),
  sortOrder: int('sort_order').notNull(), // ordering within block
}, (table) => [
  index('session_prescriptions_block_idx').on(table.blockId),
]);
```

Create Fastify plugin and integrate into application:

**plugins/sessions.ts:**
```typescript
import fp from 'fastify-plugin';
import { SessionGeneratorService } from '../modules/sessions/service';
import { sessionRoutes } from '../modules/sessions/routes';

export const sessionsPlugin = fp(async (fastify) => {
  const sessionService = new SessionGeneratorService(fastify.db);

  // Decorate fastify with session service
  fastify.decorate('sessionService', sessionService);

  // Register routes under /sessions prefix
  fastify.register(sessionRoutes, { prefix: '/sessions' });
});
```
