Script setup:
```typescript
import { computed, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import WeekCarousel from '../components/WeekCarousel.vue'
import BlockList from '../components/BlockList.vue'
import StartSessionButton from '../components/StartSessionButton.vue'
import { useWeekStore } from '../stores/weekStore'
import { useWeekData } from '../composables/useWeekData'
import { useDateNavigation } from '../composables/useDateNavigation'

const router = useRouter()
const store = useWeekStore()
const { fetchWeekSessions, loading } = useWeekData()
const { isToday } = useDateNavigation()

// Computed
const currentWeek = computed(() => store.weekDays[0]?.session?.week ?? '--')
const weekDateRange = computed(() => {
  if (store.weekDays.length < 7) return ''
  const start = store.weekDays[0].date
  const end = store.weekDays[6].date
  return `${formatShortDate(start)} - ${formatShortDate(end)}`
})

const selectedDayBlocks = computed(() => {
  const day = store.selectedDay
  return day?.session?.blocks ?? []
})

const showStartButton = computed(() => {
  const selected = store.selectedDate
  return selected && isToday(selected) && !isTodayCompleted.value
})

const isTodayCompleted = computed(() => {
  const day = store.selectedDay
  return day?.state === 'completed'
})

// Actions
function onStartSession() {
  // Navigate to Day Player (Phase 7)
  // For now, just log or show toast
  router.push({ name: 'day-player', params: { date: store.selectedDate } })
}

// Lifecycle
onMounted(async () => {
  await fetchWeekSessions()
})
```
