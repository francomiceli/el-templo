       async getOutstandingBalances(
         filters: OutstandingBalancesFilters,
         scope: { isOwner: boolean }
       ): Promise<OutstandingBalancesResult> {
         const page = filters.page ?? 1;
         const limit = filters.limit ?? 50;
         const offset = (page - 1) * limit;

         // Build base WHERE: balances.amount > 0
         const conds: SQL[] = [gt(schema.balances.amount, 0)];
         if (filters.branchId !== undefined) {
           // branchId filter implicitly excludes debt_balance rows (no branch)
           conds.push(eq(schema.subscriptions.branchId, filters.branchId));
         }
         if (filters.country !== undefined) {
           // Non-owner scope: only own country. Same constraint excludes
           // debt_balance rows lacking a branch — acceptable per D-04 (debt_balance is rare).
           conds.push(eq(schema.branches.country, filters.country));
         }
         if (filters.currency !== undefined) {
           conds.push(eq(schema.balances.currency, filters.currency));
         }
         if (filters.search !== undefined && filters.search.trim().length > 0) {
           const term = `%${filters.search.trim()}%`;
           conds.push(
             or(
               like(schema.users.firstName, term),
               like(schema.users.lastName, term),
             )!
           );
         }

         // Main query: balances LEFT JOIN subscriptions LEFT JOIN plans LEFT JOIN branches LEFT JOIN users
         // ORDER BY effective_date ASC (== ageInDays DESC, since older effectiveDate → larger ageInDays).
         // LIMIT/OFFSET for pagination.
         const rows = await this.db
           .select({...})
           .from(schema.balances)
           .leftJoin(schema.subscriptions, /* targetKind='subscription' AND targetId=subscriptions.id */)
           .leftJoin(schema.subscriptionPlans, /* */)
           .leftJoin(schema.branches, /* */)
           .leftJoin(schema.users, eq(schema.users.id, schema.balances.memberId))
           .where(and(...conds))
           .orderBy(asc(/* effectiveDate computed via COALESCE(subscriptions.startDate, balances.createdAt::date) */))
           .limit(limit)
           .offset(offset);

         // Compute ageInDays + bucket per row in JS (same as getOutstandingConcepts).
         const mapped: OutstandingBalanceRow[] = rows.map(r => {
           const effectiveDate = /* COALESCE logic */;
           const ageInDays = computeAgeInDays(effectiveDate);
           const bucket = computeBucket(ageInDays);
           return { ... };
         });

         // Sort by ageInDays DESC after JS computation (ORDER BY effectiveDate ASC produces same order, but explicit sort guards against COALESCE quirks).
         mapped.sort((a, b) => b.ageInDays - a.ageInDays);

         // Total count (separate query, no LIMIT)
         const [countRow] = await this.db
           .select({ count: sql<number>`COUNT(*)` })
           .from(schema.balances)
           .leftJoin(/* same joins */)
           .where(and(...conds));
         const total = Number(countRow?.count ?? 0);

         // bucketTotals — aggregate across full filtered set (no LIMIT)
         const totalsRows = await this.db
           .select({
             currency: schema.balances.currency,
             effectiveDate: /* COALESCE */,
             amount: schema.balances.amount,
           })
           .from(schema.balances)
           .leftJoin(/* same joins */)
           .where(and(...conds));

         let bucketTotals: BucketTotals | Record<string, BucketTotals>;
         if (scope.isOwner) {
           // Multi-currency keyed map
           const map: Record<string, BucketTotals> = {};
           for (const r of totalsRows) {
             const bucket = computeBucket(computeAgeInDays(r.effectiveDate));
             const key = r.currency;
             if (!map[key]) map[key] = { '0-30': 0, '31-60': 0, '61-90': 0, '90+': 0 };
             map[key][bucket] += Number(r.amount);
           }
           bucketTotals = map;
         } else {
           // Flat (single currency expected via country scope)
           const flat: BucketTotals = { '0-30': 0, '31-60': 0, '61-90': 0, '90+': 0 };
           for (const r of totalsRows) {
             const bucket = computeBucket(computeAgeInDays(r.effectiveDate));
             flat[bucket] += Number(r.amount);
           }
           bucketTotals = flat;
         }

         return { rows: mapped, total, page, limit, bucketTotals };
       }

       function computeAgeInDays(effectiveDate: string): number {
         const today = new Date(); today.setHours(0,0,0,0);
         const eff = new Date(effectiveDate + "T00:00:00");
         return Math.max(0, Math.floor((today.getTime() - eff.getTime()) / MS_PER_DAY));

         if (ageInDays <= 60) return '31-60';
         if (ageInDays <= 90) return '61-90';
