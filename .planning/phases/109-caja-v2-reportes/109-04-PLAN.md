       import { formatPrice } from 'src/utils/format-price';

         DebtBucket,
         OutstandingBalanceRow,
         BucketTotals,
         OutstandingBalancesFilters,
         OutstandingBalancesResult,
       } from 'src/types/transaction';

       interface Props {
         branchOptions: Array<{ id: number; name: string }>;
         displayCurrency: 'ARS' | 'EUR';
         countryScope: 'AR' | 'ES' | undefined;
         isOwner: boolean;
       }
       const props = defineProps<Props>();

       async function onExport(): Promise<void> {
         exporting.value = true;
         try {
           const all: OutstandingBalanceRow[] = [];
           let page = 1;
           // Loop until we have `total` rows, with hard cap.
           // First call also tells us `total`.

             const res: OutstandingBalancesResult = await api.getOutstandingBalances({
               ...currentFilters(),
               page,
               limit: EXPORT_PAGE_SIZE,
             });
             all.push(...res.rows);
             if (all.length >= res.total || res.rows.length === 0) break;
             page += 1;
             if (all.length >= EXPORT_HARD_CAP_ROWS) {
               log.warn('Deudas export hit hard cap, truncating', {
                 cap: EXPORT_HARD_CAP_ROWS,
                 total: res.total,
               });
               break;
             }
           }
           const rows = all.map((r) => ({
             Miembro: r.memberName,
             'Plan/Concepto': r.conceptLabel,
             Sucursal: r.branchName ?? '',
             Monto: r.amount,
             Moneda: r.currency,
             'Antigüedad (días)': r.ageInDays,
             Bucket: BUCKET_LABELS_ES[r.bucket],
             'Fecha devengo': formatDate(r.effectiveDate),
             Tipo: r.targetKind,
           }));
           const ws = XLSX.utils.json_to_sheet(rows);
           const wb = XLSX.utils.book_new();
           XLSX.utils.book_append_sheet(wb, ws, 'Deudas');
           const blob = new Blob(
             [XLSX.write(wb, { type: 'array', bookType: 'xlsx' })],
             { type: 'application/octet-stream' }
           );
           const today = new Date().toISOString().slice(0, 10);
           downloadBlob(blob, `deudas-${today}.xlsx`);
