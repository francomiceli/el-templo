       ```typescript
       const TARGET_KIND_LABEL_ES: Record<string, string> = {
         subscription: 'Plan',
         debt_balance: 'Saldo',
         transaction: 'Transacción',
       };
       function conceptoCell(linkSummary: Array<{ targetKind: string; targetId: number }> | undefined): string {
         if (!linkSummary || linkSummary.length === 0) return '';
         return linkSummary
           .map((l) => `${TARGET_KIND_LABEL_ES[l.targetKind] ?? l.targetKind} #${l.targetId}`)
           .join(', ');
       }

       const EXPORT_PAGE_SIZE = 200;       // matches backend listTransactionsSchema max (D-22)
       const EXPORT_HARD_CAP_ROWS = 10_000; // 50 pages × 200 rows

       async function onExportCaja(): Promise<void> {
         const today = new Date().toISOString().slice(0, 10);
         try {
           const all: Transaction[] = [];
           let page = 1;
           // Loop until we have `total` rows, with hard cap.
           while (true) {
             const result = await transactionsApi.getList({
               ...currentFilters(),
               page,
               pageSize: EXPORT_PAGE_SIZE,
             });
             all.push(...result.items);
             if (all.length >= result.total || result.items.length === 0) break;
             page += 1;
             if (all.length >= EXPORT_HARD_CAP_ROWS) {
               log.warn('Caja export hit hard cap, truncating', {
                 cap: EXPORT_HARD_CAP_ROWS,
                 total: result.total,
               });
               break;
             }
           }
           const rows = all.map((r) => ({
             Fecha: formatDate(r.transactionDate),
             Tipo: KIND_LABELS_ES[r.kind as TransactionKind] ?? r.kind,
             'Monto total': r.amount,
             Moneda: r.currency,
             'Método de pago': PAYMENT_METHOD_LABELS_ES[r.paymentMethod] ?? r.paymentMethod,
             Sucursal: r.branchName ?? '',
             Miembro: r.memberName ?? '',
             Conceptos: conceptoCell(r.linkSummary),
             Notas: r.notes ?? '',
             Anulado: r.voidedAt ? 'Sí' : 'No',
             'Razón anulación': r.voidReason ?? '',
           }));
           const ws = XLSX.utils.json_to_sheet(rows);
           const wb = XLSX.utils.book_new();
           XLSX.utils.book_append_sheet(wb, ws, 'Caja');
           const blob = new Blob(
             [XLSX.write(wb, { type: 'array', bookType: 'xlsx' })],
             { type: 'application/octet-stream' }
           );
           downloadBlob(blob, `caja-${today}.xlsx`);
         } catch (err: unknown) {
           const message = err instanceof Error ? err.message : 'unknown error';
           log.error('Caja export failed', { error: message });
         }
       }
       ```
