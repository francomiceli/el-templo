    ```yaml
    web-check:
      name: Web - Lint & Build
      runs-on: ubuntu-latest
      defaults:
        run:
          working-directory: el-templo-web
      steps:
        - Checkout code (actions/checkout@v4)
        - Setup pnpm (pnpm/action-setup@v4, version: 10)
        - Setup Node.js (actions/setup-node@v4, node-version from env, cache pnpm, cache-dependency-path: el-templo-web/pnpm-lock.yaml)

        - Lint (pnpm run lint, continue-on-error: true)
        - Security audit (pnpm audit --audit-level=high || true)

    ```

         name: Build Web
         needs: detect-changes
         if: github.event_name == 'workflow_dispatch' || needs.detect-changes.outputs.web == 'true'
         runs-on: ubuntu-latest
         defaults:
           run:
             working-directory: el-templo-web
         steps:
           - Checkout code
           - Setup pnpm (version: 10)
           - Setup Node.js (node-version: ${{ env.NODE_VERSION }}, cache pnpm, cache-dependency-path: el-templo-web/pnpm-lock.yaml)
           - Install dependencies (pnpm install --frozen-lockfile)
           - Build for production:
             run: pnpm run generate
             env:
               NUXT_PUBLIC_API_URL: ${{ secrets.VITE_API_URL }}
               NUXT_PUBLIC_SENTRY_DSN: ${{ secrets.VITE_SENTRY_DSN }}
           - Upload Web build artifact:
             uses: actions/upload-artifact@v4
             with:
               name: web-dist
               path: el-templo-web/.output/public

       Note: Reuse existing VITE_SENTRY_DSN secret (shared DSN per user decision). NUXT_PUBLIC_API_URL reuses existing VITE_API_URL secret value.

  <action>
    1. Create `deploy/nginx/eltemplo.org` — Production landing page:
       ```nginx
       # Landing Page (SSG - static HTML per route)
       # Location: /etc/nginx/sites-available/eltemplo.org
       # NOTE: Unlike SPA apps, SSG has real HTML files for each route.
       # try_files serves the pre-rendered HTML, falling back to index.html for any dynamic routes.
       # SSL will be added by certbot --nginx

       server {
           listen 80;
           listen [::]:80;
           server_name eltemplo.org;

           root /var/www/el-templo-web;
           index index.html;

           # SSG: serve pre-rendered HTML files, fallback to index.html
           location / {
               try_files $uri $uri/index.html $uri.html /index.html =404;
           }

           # Cache static assets aggressively (Nuxt/Vite adds content hashes)
           location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot|webp|avif)$ {
               expires 1y;
               add_header Cache-Control "public, immutable";
           }

           # Security headers
           add_header X-Frame-Options "SAMEORIGIN" always;
           add_header X-Content-Type-Options "nosniff" always;
           add_header X-XSS-Protection "1; mode=block" always;
           add_header Referrer-Policy "strict-origin-when-cross-origin" always;
       }
       ```

    2. Create `deploy/nginx/web-staging.eltemplo.org` — Staging:
       ```nginx
       # Staging Landing Page (SSG)
       # Location: /etc/nginx/sites-available/web-staging.eltemplo.org
       # SSL will be added by certbot --nginx

       server {
           listen 80;
           listen [::]:80;
           server_name web-staging.eltemplo.org;

           root /var/www/staging/el-templo-web;
           index index.html;

           location / {
               try_files $uri $uri/index.html $uri.html /index.html =404;
           }

           location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot|webp|avif)$ {
               expires 1y;
               add_header Cache-Control "public, immutable";

           add_header X-XSS-Protection "1; mode=block" always;
           add_header Referrer-Policy "strict-origin-when-cross-origin" always;
       }
       ```

    3. Create `deploy/nginx/www-redirect.eltemplo.org` — www redirect:

       # www to root domain redirect
