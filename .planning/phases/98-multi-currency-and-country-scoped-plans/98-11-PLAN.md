```typescript
export async function seedESBranch(app: FastifyInstance): Promise<number> {
  // Insert branches row with country='ES', return id
  const [inserted] = await app.db
    .insert(schema.branches)
    .values({
      name: "BCN Test",
      code: "BCN-TEST",
      country: "ES",
      maxCapacity: 20,
    })
    .$returningId();
  return inserted.id;
}

export async function seedPlan(
  app: FastifyInstance,
  country: "AR" | "ES",
  name: string,
  priceRegular: number,
): Promise<number> {
  const currency = country === "ES" ? "EUR" : "ARS";
  const [inserted] = await app.db
    .insert(schema.subscriptionPlans)
    .values({
      name,
      country,
      currency,
      priceRegular,
      priceZero: priceRegular,
      planTier: "flex",
      bookingMode: "flexible",
      durationDays: 30,
      classesPerWeek: 3,
    })
    .$returningId();
  return inserted.id;
}
```

```typescript
import { describe, it, expect, beforeAll, afterAll, beforeEach } from "vitest";
import type { FastifyInstance } from "fastify";
import {
  createTestApp,
  registerUser,
  getAuthToken,
  cleanAllTestData,
  seedESBranch,
  seedPlan,
} from "./helpers";

describe("Country scope — RBAC and cross-country validation (Phase 98)", () => {
  let app: FastifyInstance;
  let arAdminToken: string;
  let esAdminToken: string;
  let ownerToken: string;
  let arMemberToken: string;
  let arPlanId: number;
  let esPlanId: number;
  let arBranchId: number;
  let esBranchId: number;
  let arMemberId: number;
  let esMemberId: number;

  beforeAll(async () => {
    app = await createTestApp();
    await cleanAllTestData();
    // AR branch is expected to already be seeded id=1; ES branch created here
    arBranchId = 1;
    esBranchId = await seedESBranch(app);
    arPlanId = await seedPlan(app, "AR", "Test AR Plan", 15000);
    esPlanId = await seedPlan(app, "ES", "Test ES Plan", 7000);

    // Create users with role + branchId
    ({ userId: arMemberId } = await registerUser({
      email: "armember@test",
      branchId: arBranchId,
      role: "member",
    }));
    ({ userId: esMemberId } = await registerUser({
      email: "esmember@test",
      branchId: esBranchId,
      role: "member",
    }));
    arAdminToken = await getAuthToken({
      email: "aradmin@test",
      branchId: arBranchId,
      role: "admin",
    });
    esAdminToken = await getAuthToken({
      email: "esadmin@test",
      branchId: esBranchId,
      role: "admin",
    });
    ownerToken = await getAuthToken({
      email: "owner@test",
      branchId: arBranchId,
      role: "owner",
    });
    arMemberToken = await getAuthToken({
      email: "armember@test",
      branchId: arBranchId,
      role: "member",
    });

    // Seed an ARS subscription for the AR member so the /me/subscription forward-compat test has something to read
    await app.inject({
      method: "POST",
      url: `/api/admin/subscriptions/members/${arMemberId}/subscription/assign`,
      headers: { authorization: `Bearer ${ownerToken}` },
      payload: {
        planId: arPlanId,
        paymentMethod: "cash",
        branchId: arBranchId,
      },
    });
  });

  afterAll(async () => {
    await cleanAllTestData();
    await app.close();
  });

  // --- REQ-98-05: server-side country scoping for non-owners ---

  it("AR admin GET /admin/subscriptions/plans returns only AR plans", async () => {
    const res = await app.inject({
      method: "GET",
      url: "/api/admin/subscriptions/plans",
      headers: { authorization: `Bearer ${arAdminToken}` },
    });
    expect(res.statusCode).toBe(200);
    const body = res.json() as {
      plans: Array<{ country: string; currency: string }>;
    };
    expect(body.plans.length).toBeGreaterThan(0);
    expect(body.plans.every((p) => p.country === "AR")).toBe(true);
    expect(body.plans.every((p) => p.currency === "ARS")).toBe(true);
  });

  it("ES admin GET /admin/subscriptions/plans returns only ES plans", async () => {
    const res = await app.inject({
      method: "GET",
      url: "/api/admin/subscriptions/plans",
      headers: { authorization: `Bearer ${esAdminToken}` },
    });
    expect(res.statusCode).toBe(200);
    const body = res.json() as {
      plans: Array<{ country: string; currency: string }>;
    };
    expect(body.plans.every((p) => p.country === "ES")).toBe(true);
    expect(body.plans.every((p) => p.currency === "EUR")).toBe(true);
  });

  it("Owner GET /admin/subscriptions/plans?country=ES returns only ES plans", async () => {
    const res = await app.inject({
      method: "GET",
      url: "/api/admin/subscriptions/plans?country=ES",
      headers: { authorization: `Bearer ${ownerToken}` },
    });
    expect(res.statusCode).toBe(200);
    const body = res.json() as { plans: Array<{ country: string }> };
    expect(body.plans.every((p) => p.country === "ES")).toBe(true);
  });

  // --- REQ-98-06: cross-country write validation ---

  it("POST assignPlan with AR plan to ES-branch member returns 400 with Spanish error", async () => {
    const res = await app.inject({
      method: "POST",
      url: `/api/admin/subscriptions/members/${esMemberId}/subscription/assign`,
      headers: { authorization: `Bearer ${ownerToken}` },
      payload: {
        planId: arPlanId,
        paymentMethod: "cash",
        branchId: esBranchId,
      },
    });
    expect(res.statusCode).toBe(400);
    const body = res.json() as { message: string };
    expect(body.message).toContain("El plan no corresponde");
  });

  it("POST recordPayment with EUR on ARS subscription returns 400", async () => {
    // Find the AR member's existing subscription (created in beforeAll)
    const subsRes = await app.inject({
      method: "GET",
      url: `/api/admin/subscriptions/members/${arMemberId}`,
      headers: { authorization: `Bearer ${ownerToken}` },
    });
    const subsBody = subsRes.json() as {
      subscriptions: Array<{ id: number }>;
    };
    const subId = subsBody.subscriptions[0].id;

    const payRes = await app.inject({
      method: "POST",
      url: `/api/admin/payments`,
      headers: { authorization: `Bearer ${ownerToken}` },
      payload: {
        subscriptionId: subId,
        amount: 1000,
        currency: "EUR",
        paymentMethod: "cash",
      },
    });
    expect(payRes.statusCode).toBe(400);
    const body = payRes.json() as { message: string };
    expect(body.message).toContain("moneda distinta");
  });

  // --- REQ-98-11: additive-only response shape (admin side) ---

  it("GET /admin/subscriptions/plans returns country and currency on each plan row", async () => {
    const res = await app.inject({
      method: "GET",
      url: "/api/admin/subscriptions/plans?country=AR",
      headers: { authorization: `Bearer ${ownerToken}` },
    });
    const body = res.json() as {
      plans: Array<{ country: string; currency: string }>;
    };
    for (const p of body.plans) {
      expect(["AR", "ES"]).toContain(p.country);
      expect(["ARS", "EUR"]).toContain(p.currency);
    }
  });

  // --- SPEC Acceptance Criterion 7: cross-country member read forbidden ---

  it("AR admin cannot read an ES member", async () => {
    // AR admin attempts to GET the ES member's record; server must NOT return ES member data.
    // Per SPEC, server MAY return 403 (forbidden, explicit) OR 404 (hidden, treat as nonexistent).
    // Either is acceptable; leaking ES member data is a FAIL.
    const res = await app.inject({
      method: "GET",
      url: `/api/admin/subscriptions/members/${esMemberId}`, // adjust path to match actual admin get-member endpoint
      headers: { authorization: `Bearer ${arAdminToken}` },
    });
    expect([403, 404]).toContain(res.statusCode);
    // Defense-in-depth: even if status is unexpected, verify body does not leak ES member identifiers.
    const bodyText = res.body;
    expect(bodyText).not.toContain("esmember@test");
  });

  // --- AR regression ---

  it("POST assignPlan with AR plan to AR member succeeds (regression)", async () => {
    // Create a fresh AR member for this test to avoid interference with the beforeAll-seeded subscription.
    const { userId: freshArMemberId } = await registerUser({
      email: `armember-reg-${Date.now()}@test`,
      branchId: arBranchId,
      role: "member",
    });
    const res = await app.inject({
      method: "POST",
      url: `/api/admin/subscriptions/members/${freshArMemberId}/subscription/assign`,
      headers: { authorization: `Bearer ${ownerToken}` },
      payload: {
        planId: arPlanId,
        paymentMethod: "cash",
        branchId: arBranchId,
      },
    });
    expect([200, 201]).toContain(res.statusCode);
    const body = res.json() as { subscription: { currency: string } };
    expect(body.subscription.currency).toBe("ARS");
  });

  // --- REQ-98-11: member-app forward-compat ---
  // The currently-deployed member apps (iOS + Android) read these two endpoints.
  // Phase 98 must return a SUPERSET of the pre-phase-98 shape: all old keys present, no renames,
  // plus the new currency (and where applicable country) keys added additively.

  it("member plans catalog response is a superset of pre-phase-98 shape", async () => {
    const res = await app.inject({
      method: "GET",
      url: "/api/members/plans/catalog", // adjust to the actual member-facing catalog endpoint
      headers: { authorization: `Bearer ${arMemberToken}` },
    });
    expect(res.statusCode).toBe(200);
    const body = res.json() as { plans: Array<Record<string, unknown>> };
    expect(Array.isArray(body.plans)).toBe(true);
    expect(body.plans.length).toBeGreaterThan(0);

    // Enumerate pre-phase-98 keys that the deployed app reads.
    // If any of these is missing or renamed in the response, the deployed app will break.
    const preExistingKeys = [
      "id",
      "name",
      "priceRegular",
      "priceZero",
      "priceCreditCard",
      "durationDays",
      "classesPerWeek",
      "planTier",
      "bookingMode",
      "planCategory",
      "isTrial",
      "isGroup",
      "multiBranch",
    ];

    for (const plan of body.plans) {
      for (const key of preExistingKeys) {
        expect(plan).toHaveProperty(key);
      }
      // New additive keys from Phase 98:
      expect(plan).toHaveProperty("currency");
      expect(plan).toHaveProperty("country");
    }
  });

  it("member subscription response is a superset of pre-phase-98 shape", async () => {
    // AR member (seeded with an active ARS subscription in beforeAll) calls /me/subscription
    const res = await app.inject({
      method: "GET",
      url: "/api/members/me/subscription", // adjust to the actual member-facing subscription endpoint
      headers: { authorization: `Bearer ${arMemberToken}` },
    });
    expect(res.statusCode).toBe(200);
    const body = res.json() as Record<string, unknown>;

    // Pre-phase-98 subscription response keys the deployed app reads.
    // Adjust to match the ACTUAL response shape the endpoint returns today — read the current
    // handler in subscriptions/member-routes.ts to get the exact keys, then assert each is still present.
    const preExistingKeys = [
      "id",
      "userId",
      "planId",
      "startDate",
      "endDate",
      "priceRegular",
      "priceZero",
      "status",
    ];

    for (const key of preExistingKeys) {
      expect(body).toHaveProperty(key);
    }
    // New additive key from Phase 98:
    expect(body).toHaveProperty("currency");
  });
});
```
