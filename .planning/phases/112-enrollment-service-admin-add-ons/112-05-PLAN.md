    **Step B — Extend `el-templo-admin/src/composables/useProgramsApi.ts`** with the 4 methods:

    ```typescript
    import { api } from 'src/boot/axios';

    export interface EnrollmentRow {
      id: number;
      programId: number;
      programName: string;
      status: 'active' | 'paused' | 'completed' | 'expired' | 'cancelled';
      source: 'plan_linked' | 'plan_bundle' | 'admin_addon';
      pricePaid: number | null;
      assignedAt: string;
      assignedByName: string | null;
      currentWeek: number;
      durationWeeks: number;
    }

    export interface ProgramOption {
      id: number;
      name: string;
      durationWeeks: number;
    }

    export interface AssignAddonBody {
      programId: number;
      pricePaid?: number | null;
      paymentMethod?: 'cash' | 'transfer' | 'card' | 'aura_credit' | 'internal';
      notes?: string | null;
    }

    export function useProgramsApi() {
      async function listEnrollmentsForUser(userId: number): Promise<EnrollmentRow[]> {
        const { data } = await api.get<{ enrollments: EnrollmentRow[] }>(
          `/api/admin/programs/enrollments/user/${userId}`,
        );
        return data.enrollments;
      }

      async function listActivePrograms(): Promise<ProgramOption[]> {
        const { data } = await api.get<{ programs: ProgramOption[] }>(
          `/api/admin/programs`,
          { params: { isActive: true } },
        );
        return data.programs;
      }

      async function assignAddon(userId: number, body: AssignAddonBody): Promise<{ enrollmentId: number }> {
        const { data } = await api.post<{ enrollmentId: number }>(
          `/api/admin/users/${userId}/program-addons`,
          body,
        );
        return data;
      }

      async function cancelEnrollment(enrollmentId: number): Promise<void> {
        await api.post(`/api/admin/programs/enrollments/${enrollmentId}/cancel`);
      }

      return { listEnrollmentsForUser, listActivePrograms, assignAddon, cancelEnrollment };
    }
    ```

    ```vue
    import { ref, onMounted } from 'vue';
    import { useQuasar } from 'quasar';
    import { useProgramsApi, type EnrollmentRow } from 'src/composables/useProgramsApi';
    import AssignProgramAddonDialog from './AssignProgramAddonDialog.vue';

    const props = defineProps<{ userId: number }>();
    const { listEnrollmentsForUser, cancelEnrollment } = useProgramsApi();
    const $q = useQuasar();

    const enrollments = ref<EnrollmentRow[]>([]);

    const showAssignDialog = ref(false);

    async function load() {
      loading.value = true;
      try {
        enrollments.value = await listEnrollmentsForUser(props.userId);
      } catch (err: unknown) {
        $q.notify({ type: 'negative', message: 'No se pudieron cargar los programas.' });
      } finally {
        loading.value = false;
      }

    function sourceBadge(source: EnrollmentRow['source']): { label: string; color: string } {
      if (source === 'admin_addon') return { label: 'Add-on', color: 'accent' };
      return { label: 'Incluido en plan', color: 'primary' };
    }

    function statusLabel(status: EnrollmentRow['status']): string {
      const map: Record<EnrollmentRow['status'], string> = {
        active: 'Activo',
        paused: 'Pausado',
        completed: 'Completado',
        expired: 'Expirado',
        cancelled: 'Cancelado',
      };
      return map[status];
    }

    function formatPrice(p: number | null, source: EnrollmentRow['source']): string {
      if (source !== 'admin_addon') return '-';
      if (p === null) return 'Regalo';
      if (p === 0) return 'Regalo';
      return `$${p.toLocaleString('es-AR')}`;

          await load();
        } catch (err: unknown) {
          $q.notify({ type: 'negative', message: 'No se pudo cancelar la inscripcion.' });
        }
      });

    function onAssignSuccess() {
      showAssignDialog.value = false;
      load();
    }

    onMounted(load);
    ```

    ```vue
    <script setup lang="ts">
    import { ref, computed, watch } from 'vue';
    import { useQuasar } from 'quasar';
    import { useProgramsApi, type ProgramOption } from 'src/composables/useProgramsApi';
    import type { AxiosError } from 'axios';

    const props = defineProps<{
      modelValue: boolean;
      userId: number;
      existingProgramIds: number[];
    }>();
    const emit = defineEmits<{
      'update:modelValue': [value: boolean];
      success: [];
    }>();

    const { listActivePrograms, assignAddon } = useProgramsApi();
    const $q = useQuasar();

    const show = computed({
      get: () => props.modelValue,
      set: (v) => emit('update:modelValue', v),
    });

    const allPrograms = ref<ProgramOption[]>([]);
    const selectedProgramId = ref<number | null>(null);
    const pricePaid = ref<number>(0);
    const notes = ref<string>('');
    const submitting = ref(false);

    const availablePrograms = computed(() =>
      allPrograms.value.filter((p) => !props.existingProgramIds.includes(p.id)),
    );

    async function loadPrograms() {
      try {
        allPrograms.value = await listActivePrograms();
      } catch (err: unknown) {
        $q.notify({ type: 'negative', message: 'No se pudieron cargar los programas.' });
      }
    }

    function reset() {
      selectedProgramId.value = null;
      pricePaid.value = 0;
      notes.value = '';
    }

    function mapBackendError(code: string | undefined, fallback: string): string {
      const map: Record<string, string> = {
        ASSIGN_PLAN_FIRST: 'Asigna un plan al miembro antes de agregar un programa adicional.',
        PROGRAM_ALREADY_ACTIVE: 'El miembro ya esta inscripto en este programa. Cancela la inscripcion existente primero.',
      };
      return code && map[code] ? map[code] : fallback;
    }

    async function onSubmit() {
      if (!selectedProgramId.value) return;
      submitting.value = true;
      try {
        await assignAddon(props.userId, {
          programId: selectedProgramId.value,
          pricePaid: pricePaid.value,
          notes: notes.value || null,
        });
        $q.notify({ type: 'positive', message: 'Programa asignado.' });
        reset();
        emit('success');
      } catch (err: unknown) {
        const axiosErr = err as AxiosError<{ code?: string; message?: string }>;
        const status = axiosErr.response?.status;
        const code = axiosErr.response?.data?.code;
        const fallback =
          status === 403
            ? 'No tienes permisos para asignar programas.'
            : status === 404
              ? 'Programa no encontrado o inactivo.'
              : 'Error inesperado, intenta nuevamente.';
        $q.notify({ type: 'negative', message: mapBackendError(code, fallback) });
      } finally {
        submitting.value = false;
      }
    }

    watch(show, (v) => {
      if (v) {
        loadPrograms();
        reset();
      }
    });
    </script>

    <template>
      <q-dialog v-model="show" persistent>
        <q-card style="min-width: 420px;">
          <q-card-section>
            <div class="text-h6">Asignar programa adicional</div>
          </q-card-section>
          <q-card-section>
            <q-form @submit.prevent="onSubmit" class="q-gutter-md">
              <q-select
                v-model="selectedProgramId"
                :options="availablePrograms"
                option-value="id"
                option-label="name"
                emit-value
                map-options
                label="Programa"
                outlined
                :rules="[(v) => v !== null || 'Selecciona un programa']"
              />
              <q-input
                v-model.number="pricePaid"
                type="number"
                min="0"
                step="1"
                label="Precio pagado (0 = regalo)"
                outlined
                hint="Se cobra en la moneda del plan activo del miembro"
              />
              <q-input
                v-model="notes"
                type="textarea"
                label="Notas (opcional)"
                outlined
                rows="2"
                maxlength="500"
              />
              <div class="row q-gutter-sm justify-end">
                <q-btn flat label="Cancelar" v-close-popup />
                <q-btn
                  color="primary"
                  label="Asignar"
                  type="submit"
                  :loading="submitting"
                  :disable="!selectedProgramId"
                />
              </div>
            </q-form>
          </q-card-section>
        </q-card>
      </q-dialog>
    </template>
    ```
