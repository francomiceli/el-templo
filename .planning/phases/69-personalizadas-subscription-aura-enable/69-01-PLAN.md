    **1. Add `checkSubscription` to PersonalizadasService** (`el-templo-api/src/modules/personalizadas/service.ts`):

    Add a new method to the `PersonalizadasService` class:
    ```typescript
    /**
     * Check that the member has an active subscription with a personalizada-enabled plan.
     * Throws an error with a 403-appropriate message if not.
     */
    async checkSubscription(userId: number): Promise<void> {
      // Query for active/paused subscription joined to plan where isPersonalizada = true
      const [sub] = await this.db
        .select({ id: schema.subscriptions.id })
        .from(schema.subscriptions)
        .innerJoin(
          schema.subscriptionPlans,
          eq(schema.subscriptionPlans.id, schema.subscriptions.planId),
        )
        .where(
          and(
            eq(schema.subscriptions.userId, userId),
            or(
              eq(schema.subscriptions.status, "active"),
              eq(schema.subscriptions.status, "paused"),
            ),
            eq(schema.subscriptionPlans.isPersonalizada, true),
          ),
        )
        .limit(1);

      if (!sub) {
        throw new SubscriptionRequiredError();
      }
    }
    ```
