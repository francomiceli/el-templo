```typescript
export function getBlockHeaderGradient(role: BlockRole): string {
  const isTerracotta = role === "NUCLEUS" || role === "DEUTEROS_1";
  if (isTerracotta) {
    return `linear-gradient(135deg, ${BRAND_TERRACOTTA} 0%, #946b52 100%)`;
  }
  return `linear-gradient(135deg, ${BRAND_AGED_GOLD} 0%, #8f6b53 100%)`;
}
```

This already handles ROM roles correctly since they fall through to the aged gold gradient. But the TypeScript compiler needs the Record to be exhaustive if using a Record type. Since the function uses if/else (not a Record), ROM roles naturally fall through. No change needed to the function body, just ensure the type annotation is correct.
