```ts
import { describe, it, expect } from "vitest";
import { getSystemPrompt } from "../../src/ai/system-prompt.js";
import { getBusinessKnowledge } from "../../src/ai/knowledge.js";
import { BASELINE_CHARS } from "../fixtures/pb1-e1a-baseline.js";

describe("Prompt size regression — PB1.E1A lead (KGATE-05)", () => {
  it("rendered PB1.E1A prompt is at least 20% smaller than pre-refactor baseline (revised threshold)", () => {
    const rendered = getSystemPrompt({
      clientState: "lead",
      activePlaybook: "PB1",
      currentStage: "E1A",
    });
    const maxAllowed = Math.floor(BASELINE_CHARS * 0.8);
    expect(rendered.length).toBeLessThanOrEqual(maxAllowed);
  });

  it("knowledge block alone is ≥35% smaller for leads (structural goal)", () => {
    const full = getBusinessKnowledge();
    const lead = getBusinessKnowledge("lead");
    expect(lead.length).toBeLessThanOrEqual(Math.floor(full.length * 0.65));
  });

  it("non-lead states do not shrink (zero regression)", () => {
    // Trial state + PB2.entry (or any non-lead playbook) should stay near the
    // baseline size — we are NOT asserting exact equality because playbook
    // stage content differs between PB1.E1A and PB2, but we DO assert that
    // the knowledge block is un-gated for non-leads by checking the rendered
    // prompt contains retention content that the lead output excludes.
    const trialRendered = getSystemPrompt({ clientState: "trial" });
    expect(trialRendered).toContain("Estrategias de Retencion");
    expect(trialRendered).toContain("Ayuda con la App");
  });
});
```

```ts
import { describe, it, expect } from "vitest";
import { getBusinessKnowledge } from "../src/ai/knowledge.js";
import type { ClientState } from "../src/state/machine.js";

describe("Knowledge gating — per ClientState", () => {
  const full = getBusinessKnowledge();

  describe("lead (KGATE-02)", () => {
    const lead = getBusinessKnowledge("lead");

    it("includes discovery-relevant content", () => {
      expect(lead).toContain("Que es El Templo");
      expect(lead).toContain("Planes Flex"); // base pricing rendered
      expect(lead).toContain("Reglas Zero"); // or "Precios Zero" — use whatever the section heading actually emits
      expect(lead).toContain("Horarios por Sede");
      expect(lead).toContain("Clase de Prueba");
      expect(lead).toContain("Tecnicas de Venta");
      expect(lead).toContain("Reglas de Oro");
      // Sales objections (kept under 'Objeciones de venta')
      expect(lead).toContain("Es caro");
      expect(lead).toContain("No tengo tiempo");
    });

    it("excludes member-only content", () => {
      expect(lead).not.toContain("Estrategias de Retencion");
      expect(lead).not.toContain("Ayuda con la App");
      expect(lead).not.toContain("Politicas del Centro");
      // Retention objection "No me convencio / tengo dudas" is now in the
      // retention-tagged objections section and must NOT appear for leads.
      expect(lead).not.toContain("No me convencio");
      // Upgrade paths block
      expect(lead).not.toContain("Caminos de mejora de plan");
    });

    it("is smaller than the full set", () => {
      expect(lead.length).toBeLessThan(full.length);
    });
  });

  describe("non-lead states (KGATE-03)", () => {
    const nonLead: Exclude<ClientState, "lead">[] = [
      "trial",
      "active_member",
      "inactive_member",
      "expired_member",
    ];

    for (const state of nonLead) {
      it(`${state} receives the full knowledge set (identical to no-arg call)`, () => {
        expect(getBusinessKnowledge(state)).toBe(full);
      });
    }
  });

  describe("backward compat (KGATE-04)", () => {
    it("no-arg call returns full set including retention + app help", () => {
      expect(full).toContain("Estrategias de Retencion");
      expect(full).toContain("Ayuda con la App");
      expect(full).toContain("Caminos de mejora de plan");
      expect(full).toContain("No me convencio");
    });

    it("explicit undefined is equivalent to no-arg", () => {
      expect(getBusinessKnowledge(undefined)).toBe(full);
    });
  });
});
```

Notes:

- Literal strings used in `toContain` must match what the refactored knowledge.ts actually emits. If Plan 02 chose slightly different heading text (e.g., `*Objeciones de venta*` vs a different spelling), adjust the assertions to match the actual output rather than editing the source. DO NOT change knowledge.ts in this plan.
- No `any`. Use the imported `ClientState` type for exhaustive iteration.
- `const full = getBusinessKnowledge()` is computed once at suite load — this is a pure function so it's safe.
