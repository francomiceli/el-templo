    return FAIL_HARD("NO_FORMAT_CANDIDATES", block_ctx) 
 
  # Selección determinista: recommended_rank -> simplicity_rank -> 
format_id 
  ordered = sort_by_tie_breakers(candidates, 
config.format_tie_breakers) 
  chosen = ordered[0] 
 
  instance = instantiate_format_params(chosen, block_ctx, 
config.format_param_rules) 
 
  return (instance, Trace(candidates=candidates.ids, 
chosen=chosen.id, instance=instance)) 
 
 
42.9 Stage 6 — pick_exercises (core + fallback P40) 
42.9.1 Construcción de requisitos por slot 
function build_slot_requirements(block_ctx, slot_contraction: 
Contraction) -> Requirements: 
  return Requirements( 
    route = block_ctx.route, 
    pattern = block_ctx.pattern, 
    level_group = block_ctx.level_group,           # traduce a 
level_code(s) permitidos 
    target_level_num = block_ctx.target_level_num, # si aplica 
    contraction = slot_contraction, 
    difficulty_bucket = block_ctx.profile.difficulty_bucket, 
    category_scope = block_ctx.category_scope, 
    equipment = block_ctx.equipment_constraints 
  ) 
 
42.9.2 Filtrado base 
function filter_exercises(exdb, req: Requirements) -> 
list<Exercise>: 
  return exdb 
    .where(e.route == req.route) 
    .where(e.pattern == req.pattern) 
    .where(level_match(e, req.level_group, req.target_level_num)) 
    .where(e.contraction == req.contraction) 
    .where(e.difficulty_relative in 
allowed_difficulty(req.difficulty_bucket)) 
    .where(category_in_scope(e, req.category_scope)) 
    .where(equipment_ok(e, req.equipment)) 
 
42.9.3 Resolver insuficiencia (fallback ladder) 
function resolve_candidates_with_fallback(exdb, req, config) -> 
(list<Exercise>, FallbackTrace): 
  tier = 0 
  cur_req = req 
  while tier <= config.fallback_policy.max_tier: 
    pool = filter_exercises(exdb, cur_req) 
    if pool.size > 0: 
      return (stable_sort(pool), FallbackTrace(tier=tier, 
req=cur_req, status="OK")) 
 
    # aplicar siguiente relajación según tipo de falla 
    failure_code = classify_failure(exdb, cur_req)   # 
NO_MATCH_CONTRACTION, NO_MATCH_SCOPE, etc. 
    cur_req = relax_requirements(cur_req, failure_code, tier, 
config.fallback_policy) 
    tier += 1 
 
  return FAIL_HARD("NO_EXERCISE_CANDIDATES_AFTER_FALLBACK", {req}) 
 
42.9.4 Selección global N (cobertura + dedup + colisiones) 
function pick_exercises(block_ctx, exdb, config) -> (list<Exercise>, 
SelectionTrace): 
  slots = block_ctx.contraction_mix.slots  # len = N 
  pools = [] 
 
  for i in range(0, slots.len): 
    req = build_slot_requirements(block_ctx, slots[i]) 
    (pool_i, fb_trace_i) = resolve_candidates_with_fallback(exdb, 
req, config) 
    pools.append({slot=i, contraction=slots[i], pool=pool_i, 
fb=fb_trace_i}) 
 
  # Construcción determinista del set 
  selected = [] 
  used_ids = set() 
  used_relations = set() 
 
  for slot in pools in stable_order:   # orden: EXC primero, luego 
ISO, luego CON (si lo definís así) 
    candidate = choose_first_non_colliding(slot.pool, used_ids, 
used_relations, config.dedup_rules) 
    if candidate == null: 
      # deadlock por colisiones: fallback dedup 
      candidate = resolve_deadlock(slot, used_ids, used_relations, 
config) 
      if candidate == null: 
        return FAIL_HARD("DEDUP_COLLISION_DEADLOCK", block_ctx) 
 
    selected.append(candidate) 
    used_ids.add(candidate.exercise_id) 
    used_relations.union(candidate.relates_to) 
 
  return (selected, SelectionTrace(pools=pools.summary, 
selected_ids=selected.ids)) 
 
 
42.10 Stage 7 — prescribe (dosis + ladders + tempo + 
descanso + notas) 
function prescribe(block_ctx, chosen_exercises: list<Exercise>, T, 
config) -> (list<ExercisePrescription>, Trace): 
  N = chosen_exercises.len 
  budget = block_ctx.profile.reps_budget_total 
 
  doses = allocate_doses(budget, N, config.dose_policy)  # lista de 
dose_rep_equiv 
 
  prescriptions = [] 
  for i in 0..N-1: 
    e = chosen_exercises[i] 
    dose = doses[i] 
 
    # convertir rep_equiv → reps/seconds según contracción 
    (reps, seconds) = rep_equiv_to_units(dose, e.contraction, 
config.rep_equiv_rules) 
 
    ladder = maybe_generate_ladder(block_ctx, e, dose, 
config.ladder_rules) 
    tempo  = maybe_assign_tempo(block_ctx, e, config.tempo_rules) 
 
    rest = derive_rest(block_ctx, e, config.rest_rules) 
 
    adjustment = suggest_adjustment(block_ctx, e, 
config.adjustment_rules) # P35 
 
    notes = generate_notes(block_ctx, e, ladder, tempo, rest, 
adjustment, config.note_templates) 
 
    prescriptions.append(ExercisePrescription( 
      exercise_id=e.exercise_id, 
      contraction=e.contraction, 
      dose_rep_equiv=dose, 
      reps=reps, 
      seconds=seconds, 
      ladder=ladder, 
      tempo=tempo, 
      rest=rest, 
      adjustment=adjustment, 
      notes=notes 
    )) 
 
  return (prescriptions, Trace(doses=doses, 
format=block_ctx.format)) 
 
 
42.11 Stage 8 — assemble_block / assemble_day / 
assemble_week 
function assemble_block(block_ctx, T, config) -> BlockPlan: 
  (spom, spom_trace) = resolve_spom(block_ctx.week_id, 
block_ctx.route, T, config) 
  (profile, mix, bud_trace) = derive_budget(spom.pct, T, config) 
  (scope, scope_trace) = resolve_category_scope(spom, T, config) 
 
  block_ctx2 = block_ctx 
    .with(pct=spom.pct, pattern=spom.active_pattern, 
profile=profile, contraction_mix=mix, category_scope=scope) 
 
  (format_instance, fmt_trace) = pick_format(block_ctx2, T, config) 
 
  block_ctx3 = block_ctx2.with(format=format_instance) 
 
  (exercises, sel_trace) = pick_exercises(block_ctx3, T.exercise_db, 
config) 
  (prescriptions, pres_trace) = prescribe(block_ctx3, exercises, T, 
config) 
 
  block = BlockPlan( 
    block_id=block_ctx3.block_id, 
    role=block_ctx3.role, 
    route=block_ctx3.route, 
    pattern=block_ctx3.pattern, 
    pct=block_ctx3.pct, 
    intensity_profile=profile, 
    category_scope=scope, 
    contraction_mix=mix, 
    format=format_instance, 
    exercises=prescriptions, 
    notes=generate_block_notes(block_ctx3, config.note_templates), 
    traces=merge_traces(spom_trace, bud_trace, scope_trace, 
fmt_trace, sel_trace, pres_trace) 
  ) 
 
  validate_block_executability(block, config)  # P26 checks + null 
critical checks 
  return block 
 
function assemble_day(day_ctx, T, config) -> DayPlan: 
  blocks = [] 
  for block_ctx in day_ctx.block_contexts_in_fixed_order: 
    blocks.append(assemble_block(block_ctx, T, config)) 
 
  day = DayPlan(day_id=day_ctx.day_id, 
level_group=day_ctx.level_group, blocks=blocks, traces=...) 
  validate_day_coherence(day, config) # P37 
  return day 
 
function assemble_week(week_skeleton, T, config) -> WeekPlan: 
  days = [] 
  for (day_id, level_group) in 
week_skeleton.assignments.keys_in_stable_order: 
    day_ctx = build_day_context(week_skeleton, day_id, level_group, 
config) 
    days.append(assemble_day(day_ctx, T, config)) 
 
  week = WeekPlan(week_id=week_skeleton.week_id, days=days, 
traces=...) 
  return week

42.13 build_week_plan (top-level)

Punto 43 — J. Motor de ejecución (pipeline) 
Determinismo y reproducibilidad (seed, prioridades, tie-breakers) 
43.1 Objetivo lógico 
Garantizar que, dado el mismo input: 
 mismas tablas (SPOM, Intensidad, Contracción, Formatos, ExerciseDB, Rotador, 
etc.) 
 misma EngineConfig.version 
 mismo contexto (week_id, level_group, day_id) 
el motor produzca exactamente el mismo WeekPlan (mismos ejercicios, mismos formatos, 
mismo orden, mismas dosis), sin depender de: 
 orden físico de filas 
 “azar” 
 variaciones de runtime 
 concurrencia 
43.2 Principio: “seed” solo si hay RNG (idealmente NO hay RNG) 
Regla base: el motor debe ser determinista sin RNG. 
Entonces: 
 seed no es necesario para elegir ejercicios 
 solo podría existir como “guard rail” si en algún momento agregás sampling 
Política recomendada: 
 RNG_DISABLED = true por defecto 
 si RNG_ENABLED, toda función que haga sampling debe recibir seed explícito 
43.3 Normalización y orden estable (precondición) 
El 80% de la reproducibilidad se logra asegurando: 
1. claves canónicas (route_code, level_code, format_id, exercise_id) 
2. orden estable en toda lista intermedia 
43.3.1 Canonical IDs 
 exercise_id estable (PK) 
 format_id estable 
 spom_key = week_id + ":" + route_code 
Si una tabla no trae PK estable, el motor debe construirla (y dejar trace). 
43.3.2 Orden estable universal 
Toda colección intermedia debe estar ordenada por una key determinista: 
 candidatos de ejercicios: sort(exercises, by=exercise_id asc) 
 candidatos de formatos: sort(formats, by=(recommended_rank asc, 
format_id asc)) 
 días: Mon..Sat fijo 
 bloques: orden fijo definido por estructura (no por iteración de diccionario) 
43.4 Política de prioridades (hard vs soft constraints) 
Definir un “sistema de prioridades” explícito para que el motor siempre sepa qué sacrifica 
primero (esto evita decisiones ambiguas). 
43.4.1 Hard constraints (no se rompen salvo fallback extremo) 
Ejemplos típicos: 
 route_code del Nucleus 
 pattern_principal del día 
 level_group (alfa/delta vs sigma/omega) salvo regla definida 
 integridad de contrato (no nulls críticos) 
43.4.2 Soft constraints (se relajan según ladder P40) 
 category_scope (subir jerarquía) 
 difficulty_bucket (±1) 
 level_num (1–3) 
 contraction_mix exacta (si no hay pool) 
 dedup estricta por relates_to 
Esto debe estar codificado como: 
ConstraintPriority = [ 
HARD: route, pattern, level_group, 
SOFT: scope, difficulty_bucket, level_num, contraction_mix, dedup 
] 
43.5 Tie-breakers (criterios de desempate) — definición formal 
Siempre que haya “más de una opción válida”, se necesita desempate determinista. 
43.5.1 Tie-breakers para formatos 
Orden recomendado: 
1. recommended_rank (menor = mejor) 
2. compatibility_rank (si existe) 
3. simplicity_rank (si existe) 
4. format_id lexicográfico asc (último desempate universal) 
Formal: 
sort_key_format = (recommended_rank, compatibility_rank, 
simplicity_rank, format_id) 
pick first 
43.5.2 Tie-breakers para ejercicios (por slot) 
Orden recomendado: 
1. match exacto (si hay relajación aplicada, preferir el más cercano al target) 
 scope distance (0 exact, 1 parent, 2 grandparent…) 
 difficulty distance (0 exact, 1 off-by-1) 
 level_num distance (0 exact, 1 off-by-1) 
2. numeric_code (tu progresión interna, si existe) 
3. exercise_id lexicográfico asc 
Formal: 
sort_key_exercise = 
(scope_distance asc, 
difficulty_distance asc, 
level_num_distance asc, 
numeric_code asc, 
exercise_id asc) 
Nota: esto hace que, incluso cuando el pool es grande, la elección sea 
reproducible. 
43.5.3 Tie-breakers para selección global N (evitar colisiones) 
Cuando elegís un set de N ejercicios desde pools por slot, hay un problema de 
combinatoria. Para mantener determinismo: 
Regla: “greedy determinista con orden de slots por escasez”. 
1. ordenar slots por: 
 pool_size asc (más escaso primero) 
 contraction_priority (EXC > ISO > CON) como segundo criterio fijo 
2. dentro de cada slot, elegir el primer candidato que no colisione 
3. si hay deadlock: aplicar fallback dedup determinista (P40 Tier 1) y repetir 
Esto evita backtracking aleatorio. 
43.6 Distancia de relajación (métrica única) 
Para comparar candidatos cuando hubo fallback, necesitás una métrica única de “cercanía”. 
Definir relax_distance: 
relax_distance = 
w_scope * scope_distance + 
w_diff  * difficulty_distance + 
w_lvl   * level_num_distance + 
w_contr * contraction_substitution_distance + 
w_dedup * dedup_violation_count 
Pesos recomendados (orden, no números mágicos): 
 scope y dificultad pesan más que dedup en intensidades altas 
 contraction substitution pesa mucho si pct alto 
Esto se versiona dentro de config: 
 relax_weights_v1 
43.7 Seed (si se habilita sampling) 
Si en algún caso querés “variedad” (ej: rotación de ejercicios dentro del mismo bucket), 
entonces sí: 
43.7.1 Seed derivado determinísticamente 
No usar time(). Usar un hash de contexto: 
seed = hash(engine_version + week_id + day_id + level_group + 
block_id) 
43.7.2 Sampling reproducible 
 usar PRNG determinista (ej: Xorshift/PCG) con ese seed 
 sampling solo dentro de un pool ya ordenado 
 el sampling debe quedar traceado: 
{ 
  "rng": {"enabled": true, "seed": 19238123, "strategy": 
"pick_k_from_sorted_pool"} 
} 
 
Política: aunque uses seed, el pool debe estar ordenado y el sampling debe ser una 
transformación reproducible. 
 
43.8 Concurrencia y reproducibilidad (si paralelizás) 
Si el pipeline corre en paralelo por día/bloque, aún debe ser reproducible. 
Reglas: 
 cada bloque es función pura (no escribe en estado global) 
 
 el join final ordena por day_id, level_group, block_order 
 
 no usar estructuras no deterministas (hash maps sin orden) en outputs

44.16 Reglas de logging (cuándo loguear qué) 
44.16.1 Siempre loguear (INFO) 
 formato elegido + candidatos 
 ejercicios elegidos (ids) 
 budget derivado (reps_budget_total, N, bucket) 
44.16.2 Loguear WARNING 
 cualquier fallback tier > 0 
 pool_size < min_pool_per_slot 
 dedup relajado 
 sustitución de contracción 
 formato fallback 
44.16.3 Loguear ERROR 
 cuando el plan se genera pero viola regla “soft” fuerte: 
 distribución superior/inferior queda sesgada 
 núcleo y accesorio se pisan (si es regla de error) 
44.16.4 Loguear HARD_ERROR 
 no hay candidatos después de fallback 
 falta row en tablas source-of-truth 
44.17 Compresión / límites (para no explotar tamaño) 
Política: 
 pools: guardar size + preview (top 10 ids) + hash del conjunto si querés reproducir 
 candidates de formatos: guardar todos (son pocos) 
 events: max 50 por bloque (si más, agrupar por code) 
Ejemplo: 
"pool_hash": "sha256:..." 
44.18 Consultas típicas (qué habilita el trace) 
 “¿Por qué esta semana salió más Planche que la pasada?” 
 comparar spom_trace.pct + active_scope_key 
 “¿Por qué metiste ISO en vez de EXC?” 
 exercise_trace.slots[].fallback.actions 
 “¿Por qué eligió EMOM y no AMRAP?” 
 format_trace.candidates + tie_breakers 
 “¿Qué parte de la BD está floja?” 
 LOW_POOL por route/level/contracción agregados 
44.19 Output mínimo obligatorio (resumen) 
Para considerar el trace “auditable”, por cada bloque debe existir: 
 spom_trace.chosen.pct 
 budget_trace.derived 
 format_trace.candidates + chosen + tie_breakers 
 exercise_trace.slots[].req + pool.size + fallback 
 exercise_trace.global_selection.chosen_exercise_ids 
 prescription_trace.allocation + conversion_rules 
Si falta cualquiera → trace inválido. 
Punto 45 — K. Implementación: schemas y ejemplos 
JSON Schema completo (WeekPlan / DayPlan / BlockPlan / 
ExercisePrescription) 
Objetivo: definir un JSON Schema rígido (estilo draft 2020-12) para validar que 
el agente/emisor entrega un plan ejecutable y auditable. 
Nota: abajo va un schema completo y utilizable, pensado para validación 
automática. 
45.1 Convenciones del schema 
 additionalProperties: false en todos los objetos críticos (evita “campos 
inventados”). 
 Enums cerrados para day_id, level_group, role, pattern, contraction. 
 Campos “null permitidos” se tipan como ["null", "..."]. 
 Longitudes: exercise_prescriptions.length == 
exercise_count_target no puede forzarse perfecto en JSON Schema puro si 
exercise_count_target está en otro objeto; se valida por código (validator de 
negocio). Igual dejamos constraints parciales + el campo expected_counts para 
ayudar.

Punto 47 — K. Implementación: Suite de tests 
(validaciones unitarias por regla) 
47.1 Objetivo lógico 
Definir una suite de tests que valide el motor por reglas atómicas (unit tests), y por 
integración (pipeline completo). 
El criterio es: si una regla se rompe, el test falla con un código claro, sin ambigüedad. 
47.2 Estructura propuesta de la suite 
47.2.1 Capas de test 
1. Schema tests (sintaxis / contrato JSON) 
2. Rule unit tests (una regla → un test) 
3. Property tests (invariantes: no dependen de fixtures exactos) 
4. Integration tests (pipeline completo, snapshot) 
5. Regression tests (casos históricos: W19, etc.) 
47.2.2 Convención de naming 
test_<stage>__<rule_code>__<scenario>() 
Ej: 
 test_spom__SPOM_CONFLICT_PRIORITY__chooses_highest_priority_ro
w() 
 test_budget__BUDGET_REPS_SUM__exact_sum_matches_reps_budget() 
 test_exercises__DEDUP_RELATES_TO__no_equivalent_in_same_block(
) 
47.3 Fixtures (datos mínimos para tests) 
Para unit tests, no usar la BD entera: usar fixtures mínimos. 
47.3.1 Tipos de fixtures 
 spom_fixture_small.json 
 intensity_fixture_small.json 
 contraction_fixture_small.json 
 formats_fixture_small.json 
 exercise_db_fixture_small.json

47.3.2 Reglas de fixtures 
 Cada fixture tiene fixture_version 
 Cada tabla tiene PK estable 
 Cada set de fixtures representa un “micro-universo” controlado (10–50 rows max)
